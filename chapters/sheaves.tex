\chapter{Presheaves and sheaves}

\section{Presheaves}
We begin by giving a general definition of a presheaf.
\begin{definition}[Presheaf] \label{def:presheaf}
  Let \(X\) be a topological space and let \(\open(X)\) be its \nameref{ex:open-set-category}.
  Let \(\mathsf{C}\) be a \nameref{def:small-category}.
  The \nameref{def:category} of \(\mathsf{C}\)-valued \textbf{presheaves} is the \nameref{ex:functor-category}
  \begin{equation*}
    \PSh_{X}(\mathsf{C}) := [\open(X)^{\op}, \mathsf{C}]
  \end{equation*}
  When the topological space is clear we will write \(\PSh(\mathsf{C})\).

  The elements of this category are \namerefs{def:functor} \(\mathcal{F}: \open(X)^{\op} \to \mathsf{C}\) and morphisms are \namerefs{def:natural-transformation} \(\phi: \mathcal{F} \to \mathcal{G}\).
\end{definition}
\SetPluralName{def:presheaf}{Presheaves}

Let's rewrite \autoref{def:presheaf} less abstractly: a \(\mathsf{C}\)-valued \nameref{def:presheaf} \(\mathcal{F}\) is the collection of data
\begin{itemize}
  \item For every open set \(U \subseteq X\) an object \(\mathcal{F}(U)\) of \(\mathsf{C}\) called \textit{sections of \(\mathcal{F}\) over \(U\)}.
  \item For every inclussion of open sets \(V \subseteq U\) a morphism \(\res_{U, V}: \mathcal{F}(U) \to \mathcal{F}(V)\) in \(\mathsf{C}\) called \textit{restriction} such that
        \begin{itemize}
          \item For every open set \(U\), \(\res_{U, U} = \id_{U}\).
          \item For every inclussion of open sets \(W \subseteq V \subseteq U\), \(\res_{U, V} \circ \res_{V, W} = \res_{U, W}\).
        \end{itemize}
\end{itemize}

A \textit{morphism of \namerefs{def:presheaf}} \(\phi: \mathcal{F} \to \mathcal{G}\) is a collection of morphisms \(\{\phi(U): \mathcal{F}(U) \to \mathcal{G}(U)\}_{U \in \open(X)}\) in \(\mathsf{C}\) making the following diagram commute:
\begin{center}
  \begin{tikzcd}
    \mathcal{F}(U) \arrow[r, "\phi(U)"] \arrow[d, "\res_{U, V}"] & \mathcal{G}(U) \arrow[d, "\res_{U, V}"] \\
    \mathcal{F}(V) \arrow[r, "\phi(V)"] & \mathcal{G}(V)
  \end{tikzcd}
\end{center}



We are going to study now \namerefs{def:categorical-limit} and \namerefs{def:categorical-colimit} in \(\PSh(\mathsf{C})\).

Note that, since \(\PSh(\mathsf{C})\) is a \nameref{def:functor-category} then if \(\mathsf{C}\) has \namerefs{def:categorical-limit} or \namerefs{def:categorical-colimit} of a certain shape then \(\PSh(\mathsf{C})\) also has them and they are computed pointwise.

This means that \namerefs{def:categorical-limit} and \namerefs{def:categorical-colimit} in \(\PSh(\mathsf{C})\) can be computed \textit{open set by open set}.

\begin{proposition}
  Let \(X\) be a topological space. Let \(\mathcal{I} \to \PSh_{X}(\mathsf{C}), i \mapsto \mathcal{F}_{i}\) be a \nameref{def:diagram}.
  If \(\mathsf{C}\) has \namerefs{def:categorical-limit} and/or \namerefs{def:categorical-colimit} with shape \(\mathcal{I}\) then \(\lim_{i} \mathcal{F}_{i}\) and/or \(\colim_{i} \mathcal{F}_{i}\) exist in \(\PSh_{X}(\mathsf{C})\).
  Moreover for every open set \(U \subseteq X\)
  \begin{align*}
    \left( \lim_{i} \mathcal{F}_{i} \right)(U) = \lim_{i} \mathcal{F}_{i}(U) \qquad \left( \colim_{i} \mathcal{F}_{i} \right)(U) = \colim_{i} \mathcal{F}_{i}(U)
  \end{align*}
\end{proposition}

In particular if \(\mathsf{C}\) is an \nameref{def:abelian-category} then \namerefs{def:categorical-kernel} and \namerefs{def:categorical-cokernel} exist and are computed \textit{open set by open set}, making \(\PSh(\mathsf{C})\) an \nameref{def:abelian-category}.



\begin{remark}
  From this point onwards, we will suppose \(\mathsf{C}\) has all \namerefs{def:categorical-limit} and \namerefs{def:categorical-colimit} unless stated otherwise.
\end{remark}

\subsection{Presheaves and continuous maps}

\begin{definition}[Direct image \nameref{def:presheaf}] \label{def:direct-image-presheaf}
  Let \(f: X \to Y\) be a continuous map.
  Let \(\mathcal{F}\) be a \(\mathsf{C}\)-\nameref{def:presheaf} over \(X\).
  The \textbf{direct image \nameref{def:presheaf}} of \(\mathcal{F}\) via \(f\) is the \nameref{def:presheaf} \(f_{\ast}\mathcal{F}\) defined as, for every open subset \(V \subseteq Y\)
  \begin{equation*}
    f_{\ast}\mathcal{F}(V) = \mathcal{F}(f^{-1}(U))
  \end{equation*}
  Given \(W \subseteq V \subseteq Y\) open subsets, the restriction map comes from the commutativity of the diagram
  \begin{center}
    \begin{tikzcd}
      f_{\ast}\mathcal{F}(V) \arrow[r, equal] \arrow[d, dotted] & \mathcal{F}(f^{-1}V) \arrow[d] \\
      f_{\ast}\mathcal{F}(V) \arrow[r, equal] & \mathcal{F}(f^{-1}V)
    \end{tikzcd}
  \end{center}

  This construction defines a \nameref{def:functor} \(f_{\ast}: \PSh_{X}(\mathsf{C}) \to \PSh_{Y}(\mathsf{C})\).
\end{definition}

\begin{example}[Skyscrapper \nameref{def:presheaf}] \label{ex:skyscrapper-presheaf}
  Let \(X\) be a topological space and \(p \in X\).
  Suppose \(\mathsf{C}\) has a \nameref{def:final-object} \(\ast \in \mathsf{C}\).
  Let \(S\) be an object in \(\mathsf{C}\).
  The \textbf{skyscrapper \nameref{def:presheaf}} supported at \(S\) \(\iota_{p, \ast}\underline{S}_{\pre}\) is the \nameref{def:direct-image-presheaf} of the \nameref{ex:constant-presheaf} \(\underline{S}_{\pre}\) via the inclussion \(\iota: \{p\} \hookrightarrow X\):
  \begin{equation*}
    (\iota_{p, \ast}\underline{S}_{\pre})(U) = \begin{cases}
      S & p \in U \\
      \ast & p \notin U
    \end{cases}
  \end{equation*}
\end{example}


Using this we can construct a \textit{dual} \nameref{def:presheaf} to \autoref{def:direct-image-presheaf}.
\begin{definition}[Inverse image \nameref{def:presheaf}] \label{def:inverse-image-presheaf}
  Let \(f: X \to Y\) be a continuous map.
  Let \(\mathcal{G}\) be a \(\mathsf{C}\)-valued \nameref{def:presheaf} over \(Y\).
  The \textbf{inverse image \nameref{def:presheaf}} of \(\mathcal{G}\) via \(f\) is the \nameref{def:presheaf} \(f^{-1}\mathcal{G}\) defined as, for every open subset \(U \subseteq X\)
  \begin{equation*}
    f^{-1}\mathcal{G}(U) = \colim_{V \supseteq f(U)} \mathcal{G}(V)
  \end{equation*}
  where the \nameref{def:categorical-colimit} is taken over all the open neighbourhoods \(V\) of \(f(U)\).

  This defines a \nameref{def:functor} \(f^{-1}: \PSh_{Y}(\mathsf{C}) \to \PSh_{X}(\mathsf{C})\).
\end{definition}

\begin{remark}
  Note that the set of open neighbourhoods of \(f(U)\) is a partially ordered set via reverse inclussion.
  Therefore if \(\mathsf{C}\) is a concrete category we can describe \(f^{-1}\mathcal{G}(U)\) as \(\{(s, V) \mid f(U) \subseteq V \in \open(Y),\, s \in \mathcal{G}(V)\}\) modulo the relation \((s, V) \sim (s', V')\) if and only if there exists \(W \subseteq V \cap V'\) open with \(f(U) \subseteq W\) such that \(s|_{W} = s'|_{W}\).
\end{remark}

\begin{example}[Constant \nameref{def:presheaf}] \label{ex:constant-presheaf}
  Let \(X\) be a topological space.
  Let \(S\) be an object of \(\mathsf{C}\).
  The \textbf{constant \nameref{def:presheaf}} associated to \(S\) \(\underline{S}_{\pre}\) is the \nameref{def:inverse-image-presheaf} of \(S\) via the map \(\{\ast\} \to X\).
\end{example}

\begin{definition}[Stalk] \label{def:stalk-presheaf}
  Let \(X\) be a topological space and \(p \in X\).
  Let \(\mathcal{F}\) be a \(\mathsf{C}\)-valued \nameref{def:presheaf} over \(X\).
  The \textbf{stalk} of \(\mathcal{F}\) at \(p\) \(\mathcal{F}_{p}\) is the \nameref{def:inverse-image-presheaf} of \(\mathcal{F}\) via \(X \to \{p\}\):
  \begin{equation*}
    \mathcal{F}_{p} = \colim_{U \ni p} \mathcal{F}(U)
  \end{equation*}
\end{definition}

\begin{proposition} \label{prop:direct-inverse-image-adjunction-presheaves}
  Let \(f: X \to Y\) be a continuous map.
  We have an \nameref{def:adjunction} between the \nameref{def:direct-image-presheaf} functor and the \nameref{def:inverse-image-presheaf} \(f^{-1} \dashv f_{\ast}\).
\end{proposition}
\begin{proof}
  We define the \nameref{def:categorical-unit} \(\eta: 1 \implies f_{\ast}f^{-1}\) as the map with components given by the canonical insertion maps to the colimit
  \begin{align*}
    \mathcal{F}(W) \to \colim_{U \supseteq (f \circ f^{-1})(W)} \mathcal{F}(U) = f^{-1}\mathcal{F}(f^{-1}W) = f_{\ast}f^{-1}\mathcal{F}(W)
  \end{align*}
  Note that this maps exist because \((f \circ f^{-1})(W) \subseteq W\).

  On the other hand, note that if \(f(V) \subseteq U\) then \(V \subseteq (f^{-1} \circ f)(V) \subseteq f^{-1}(U)\), i.e: we have a restricion map \(\mathcal{G}(f^{-1}(U)) \to \mathcal{G}(V)\).
  Therefore by \hyperref[def:categorical-colimit]{the universal property of the colimit} we have a unique map
  \begin{align*}
    f^{-1}f_{\ast}\mathcal{G}(V) = \colim_{U \supseteq f(V)} \mathcal{G}(f^{-1}(U)) \to \mathcal{G}(V)
  \end{align*}
  Using this maps we obtain a \nameref{def:categorical-counit} \(\varepsilon: f^{-1}f_{\ast} \implies 1\).

  Since by construction the compositions \(\eta \circ \varepsilon\) and \(\varepsilon \circ \eta\) are the corresponding identity morphisms, we have proven the adjunction.
\end{proof}

Two important particular cases of the adjunction \autoref{prop:direct-inverse-image-adjunction-presheaves} are
\begin{corollary} \label{cor:constant-global-adjunction-presheaf}
  Let \(X\) be a topological space and \(p \in X\).
  Let \(\mathcal{F}\) be a \(\mathsf{C}\)-valued \namerefs{def:presheaf} over \(X\) and let \(A\) be an object in \(\mathsf{C}\).
  Then
  \begin{equation*}
    \Hom_{\PSh_{X}(\mathsf{C})}(\underline{A}_{\pre}, \mathcal{F}) \simeq \Hom_{\mathsf{C}}(A, \Gamma(X, \mathcal{F}))
  \end{equation*}
\end{corollary}

\begin{corollary}\label{cor:stalk-constant-adjunction-presheaf}
  Let \(X\) be a topological space and \(p \in X\).
  Let \(\mathcal{F}\) be a \(\mathsf{C}\)-valued \namerefs{def:presheaf} over \(X\) and let \(A\) be an object in \(\mathsf{C}\).
  Then
  \begin{equation*}
    \Hom_{\mathsf{C}}(\mathcal{F}_{p}, A) \simeq \Hom_{\PSh_{X}(\mathsf{C})}(\mathcal{F}, \iota_{p, \ast}\underline{A}_{\pre})
  \end{equation*}
\end{corollary}


\section{Sheaves}
We now turn our attention to a particular class of \namerefs{def:presheaf}, following \cite[\href{https://stacks.math.columbia.edu/tag/00VL}{Tag 00VL}]{stacks-project}.

Let's start by considering the case \(\mathsf{C} = \Set\):
\begin{definition}[Sheaf of sets] \label{def:sheaf-set}
  Let \(X\) be a topological space.
  A \nameref{def:presheaf} of sets \(\mathcal{F}\) is a \textbf{sheaf} if for every open set \(U\) and open covering \(\{U_{i}\}_{i \in \mathcal{I}}\) of \(U\) the diagram
  \begin{equation} \label{diagram:sheaf-condition}
    \mathcal{F}(U) \to \prod_{i \in \mathcal{I}} \mathcal{F}(U_{i}) \rightrightarrows \prod_{(i, j) \in \mathcal{I} \times \mathcal{I}} \mathcal{F}(U_{i} \cap U_{j})
  \end{equation}
  is an \nameref{def:equalizer}.

  We denote by \(\Sh_{X}(\Set)\) (or just \(\Sh\) if the context is clear) the \nameref{def:full-subcategory} of the \hyperref[def:presheaf]{category of presheaves} of sets whose objects are sheaves of sets.

  Less abstractly this means that \(\mathcal{F}\) verifies:
  \begin{enumerate}[label=SH.\arabic*), ref=SH.\arabic*]
    \item\label{axiom:sheaf-identity} \textbf{Identity}: Given two sections \(s, t \in \sheaf{F}(U)\) such that for every \(i \in \mathcal{I} \quad s|_{U_{i}} = t|_{U_{i}}\) we have that \(s = t\).
    \item\label{axiom:sheaf-gluability} \textbf{Gluability}: Given a collection of sections \(\{s_{i} \mid s_{i} \in \sheaf{F}(U_{i})\}_{i \in \mathcal{I}}\) that agree on pairwise intersections (\(\forall i, j \in \mathcal{I} \;\; s_{i}|_{U_{i} \cap U_{j}} = s_{j}|_{U_{i} \cap U_{j}}\)) there exists a section \(s \in \sheaf{F}(U)\) such that \(\forall i \in \mathcal{I} \;\; s|_{U_{i}} = s_{i}\).
  \end{enumerate}
\end{definition}
\SetPluralName{def:sheaf-set}{Sheaves of sets}

\begin{definition}[\nameref{def:category} of \namerefs{def:sheaf-set}] \label{def:sheaf-set-category}
The \nameref{def:category} \(\Sh_{X}(\Set)\) of \namerefs{def:sheaf-set} is the \nameref{def:full-subcategory} of the category of \namerefs{def:presheaf} \(\PSh_{X}(\Set)\) whose objects are \namerefs{def:sheaf-set}.
\end{definition}

Let's try to make \autoref{def:sheaf-set} more general by considering another evaluation category \(\mathsf{C}\).

Firstly, note that if \namerefs{def:categorical-product} indexed by \(\mathcal{I}\) and \(\mathcal{I} \times \mathcal{I}\) for any indexing set \(\mathcal{I}\) exist in \(\mathsf{C}\) then the diagram \eqref{diagram:sheaf-condition} makes sense.
Secondly, we have that \eqref{diagram:sheaf-condition} is an \nameref{def:equalizer} if and only if for every object \(C \in \mathsf{C}\) the diagram in \(\Set\)
\begin{equation*}
  \Hom_{\mathsf{C}}(C, \mathcal{F}(U)) \to \prod_{i \in \mathcal{I}} \Hom_{\mathsf{C}}(C, \mathcal{F}(U_{i})) \rightrightarrows \prod_{(i, j) \in \mathcal{I} \times \mathcal{I}} \Hom_{\mathsf{C}}(C, \mathcal{F}(U_{i} \cap U_{j}))
\end{equation*}
is an \nameref{def:equalizer}.

Therefore we get
\begin{definition}[Sheaf] \label{def:sheaf}
  Let \(X\) be a topological space.
  Let \(\mathsf{C}\) be an arbitrary category.
  A \(\mathsf{C}\)-valued \nameref{def:presheaf} \(\mathcal{F}\) is a \textbf{sheaf} if for all objects \(C \in \mathsf{C}\) the \nameref{def:presheaf} of sets \(\mathcal{F}_{C}\) defined as \(\mathcal{F}_{C}(U) = \Hom_{\mathsf{C}}(C, \mathcal{F}(U))\) is a \nameref{def:sheaf-set}.

  We denote by \(\Sh_{X}(\mathsf{C})\) the \nameref{def:full-subcategory} of the \hyperref[def:presheaf]{category of \(\mathsf{C}\)-valued presheaves} of sets whose objects are \(\mathsf{C}\)-valued sheaves.
\end{definition}
\SetPluralName{def:sheaf}{Sheaves}

\begin{notation}
  From now on, to distinguish a \nameref{def:presheaf} from a \nameref{def:sheaf} we will denote the former by \(\mathcal{F}, \mathcal{G}, \mathcal{H}, \ldots\) and the latter by \(\sheaf{F}, \sheaf{G}, \sheaf{H}, \ldots\)
\end{notation}

By \autoref{def:sheaf} we have an inclussion functor \(\iota: \Sh(\mathsf{C}) \hookrightarrow \PSh(\mathsf{C})\).

For \textit{sufficiently nice}\unsure{How general do we want to go?} \namerefs{def:category}, this \nameref{def:functor} admits a \hyperref[def:adjoint]{left-adjoint} \(\PSh(\mathsf{C}) \to \Sh(\mathsf{C})\).
Let's study first the case \(\mathsf{C} = \Set\).

\begin{theorem}[Sheafification] \label{thm:sheafification-set}
  There exists a \textbf{sheafification} \nameref{def:functor} \((-)^{\#}: \PSh(\Set) \to \Sh(\Set)\) that is \hyperref[def:adjunction]{left-adjoint} to the inclussion \nameref{def:functor} \(\iota: \Sh(\Set) \to \PSh(\Set)\).
\end{theorem}
\begin{proof}
  Let \(\mathcal{F}\) be a \nameref{def:presheaf} of sets over \(X\).
  For every open subset \(U \subseteq X\) we define
  \begin{align*}
    \mathcal{F}^{\#}(U) = \left\{ f_{p} \in \prod_{p \in U} \mathcal{F}_{p} \text{ verifying } (\ast) \right\}
  \end{align*}
  where \((\ast)\) is the property \textit{"for all \(p \in U\) there exists an open neighbourhood \(V \subseteq U\) of \(p\) and a section \(s \in \mathcal{F}(V)\) such that, for every \(q \in V\), \(s_{q} = f_{q}\)"}.

  Taking \(\prod_{p \in U} \mathcal{F}_{p} \to \prod_{q \in V} \mathcal{F}_{q}\) as the restriction maps we see that \(\mathcal{F}^{\#}\) is a \nameref{def:presheaf}.
  Let's see now that \(\mathcal{F}^{\#}\) is a \nameref{def:sheaf}. To do this, consider an open subset \(U \subseteq X\) with open cover \(\{U_{i}\}_{i \in \mathcal{I}}\).
  \begin{enumerate}
    \item[\ref{axiom:sheaf-identity}] Let \(s, t \in \mathcal{F}^{\#}(U)\) be two sections such that, for all \(i \in \mathcal{I} \;\; s|_{U_{i}} = t|_{U_{i}}\).

          Then since for every \(p \in U\) there exists \(i \in \mathcal{I}\) such that \(p \in U_{i}\) we have that \(s_{p} = (s|_{U_{i}})_{p} = (t|_{u_{i}})_{p} = t_{p}\).
          Therefore \(s = t\).
    \item[\ref{axiom:sheaf-gluability}] Let \(\{s_{i} \in \mathcal{F}^{\#}(U_{i})\}_{i \in \mathcal{I}}\) be a collection of sections such that \(s_{i}|_{U_{i} \cap U_{j}} = s_{j}|_{U_{i} \cap U_{j}}\) for all \(i, j \in \mathcal{I}\).

          Consider \(s \in \prod_{p \in U} \mathcal{F}_{p}\) defined as \(s_{p} = (s_{i})_{p}\) if \(p \in U_{i}\). Note that this is well-defined by property \((\ast)\).
          To see that \(s\) verifies \((\ast)\) note that for every \(p \in U_{i}\) there exists an open neighbourhood \(V \subseteq U\) of \(p\) and \(t \in \mathcal{F}(V)\) such that, for all \(q \in V\), \(t_{q} = (s_{i})|_{q} = s_{q}\).
  \end{enumerate}

  Let \(V \subseteq U \subseteq X\) be two open sets. Consider the commutative diagram
  \begin{center}
    \begin{tikzcd}
      \mathcal{F}(U) \arrow[r] \arrow[d] & \mathcal{F}^{\#}(U) \arrow[r] \arrow[d] & \prod_{p \in U} \mathcal{F}_{p} \arrow[d] \\
      \mathcal{F}(V) \arrow[r] & \mathcal{F}^{\#}(V) \arrow[r] & \prod_{q \in V} \mathcal{F}_{q}
    \end{tikzcd}
  \end{center}
  where the vertical arrows are the corresponding restriction maps.
  By chasing this diagram we get a map \(\varphi: \mathcal{F} \to \mathcal{F}^{\#}, \varphi_{U}(s) = (s_{p})_{p \in U}\).

  It only remains to show that any morphism \(\psi: \mathcal{F} \to \sheaf{G}\), where \(\sheaf{G}\) is a \nameref{def:sheaf} of sets, factors uniquely as \(\mathcal{F} \xrightarrow{\varphi} \mathcal{F}^{\#} \to \sheaf{G}\), i.e: we have to construct a morphism \(\theta: \mathcal{F}^{\#} \to \sheaf{G}\) such that \(\psi = \theta \circ \varphi\).

  Let \(U \subseteq X\) be an open subset and \(s \in \mathcal{F}^{\#}(U)\).
  For each \(p \in U\) consider an open neighbourhood \(V^{(p)} \subseteq U\) of \(p\) and \(t^{(p)} \in \mathcal{F}(\mathcal{V}^{(p)})\) such that \(t_{q}^{(p)} = s_{q}\).
  Consider now \(\psi_{V^{(p)}}(t^{(p)}) \in \sheaf{G}(V^{(p)})\).
  Note that \(\{\psi_{V^{(p)}}(t^{(p)})\}_{p \in U}\) is a compatible family of section in \(\sheaf{G}\), therefore by the \nameref{def:sheaf} axioms there exists a unique section \(t \in \sheaf{G}(U)\) such that \(t|_{V^{(p)}} = \psi_{V^{(p)}}(s^{(p)})\).
  Therefore we can define \(\theta_{U}(s) = t\).

  By construction \(\psi = \theta \circ \varphi\) and \(\theta\) commutes with the restriction maps, making it a \hyperref[def:sheaf]{morphism of sheaves}.

  To check uniqueness, let \(\theta': \mathcal{F}^{\#} \to \sheaf{G}\) be another morphism such that \(\psi = \theta' \circ \varphi\).
  Consider \(s \in \mathcal{F}^{\#}(U)\). For each \(p \in U\) choose \(V^{(p)} \subseteq U\) and \(s^{(p)} \in \mathcal{F}(V^{(p)})\) such that \(s|_{V^{(p)}} = \varphi_{V^{(p)}}(s^{(p)})\).
  Then
  \begin{equation*}
    \theta'(s)|_{V^{(p)}} = \theta'(s|_{V^{(p)}}) = \theta'(\varphi_{V^{(p)}}(s^{(p)})) = \psi_{V^{(p)}}(s^{(p)})
  \end{equation*}
  On the other hand \(\theta(s)|_{V^{(p)}} = \psi_{V^{(p)}}(s^{(p)})\) by definition.
  Therefore since \(\sheaf{G}\) is a \nameref{def:sheaf} by \ref{axiom:sheaf-identity} we have \(\theta'(f) = \theta(f)\), i.e: \(\theta' = \theta\).
\end{proof}

We can extend the construction outlined in \autoref{thm:sheafification-set} to \namerefs{def:algebraic-category}.
\begin{lemma}
  Let \((\mathsf{C}, F)\) be an \nameref{def:algebraic-category}. Let \(\mathcal{F}\) be a \(\mathsf{C}\)-valued \nameref{def:presheaf}.
  Then there exists a \(\mathsf{C}\)-valued \nameref{def:sheaf} \(\mathcal{F}^{\#}\) and a \hyperref[def:presheaf]{morphism of presheaves} \(\mathcal{F} \to \mathcal{F}^{\#}\) satisfying
  \begin{enumerate}
    \item The map \(\mathcal{F} \to \mathcal{F}^{\#}\) identifies the underlying \nameref{def:sheaf} of sets of \(\mathcal{F}^{\#}\) with the \hyperref[thm:sheafification-set]{sheafification} of the underlying \nameref{def:presheaf} of sets of \(\mathcal{F}\).
    \item For any \hyperref[def:presheaf]{morphism of presheaves} \(\mathcal{F} \to \sheaf{G}\) where \(\sheaf{G}\) is a \(\mathsf{C}\)-valued \nameref{def:sheaf} there exists a unique factorization \(\mathcal{F} \to \mathcal{F}^{\#} \to \sheaf{G}\).
  \end{enumerate}
\end{lemma}

\begin{remark}
  We will assume that \(\mathsf{C}\) is an \nameref{def:algebraic-category} in what follows.
\end{remark}

Therefore we can assign a \nameref{def:sheaf} to any \nameref{def:presheaf} using this construction.
\begin{definition}[Sheafification] \label{def:sheafification}
  Let \(\mathcal{F}\) be a \(\mathsf{C}\)-valued \nameref{def:presheaf}.
  We call the \nameref{def:sheaf} \(\mathcal{F}^{\#}\) obtained by \autoref{thm:sheafification-set} the \textbf{sheafification} of \(\mathcal{F}\).
\end{definition}


Using this adjunction we can \textit{adapt} \autoref{def:direct-image-presheaf} and \autoref{def:inverse-image-presheaf} to the case of \namerefs{def:sheaf}
\begin{definition}[Direct image \nameref{def:sheaf}] \label{def:direct-image-sheaf}
  Let \(f: X \to Y\) be a continuous map.
  Let \(\sheaf{F}\) be a \(\mathsf{C}\)-valued \nameref{def:sheaf} over \(X\).
  The \textbf{direct image \nameref{def:sheaf}} of \(\sheaf{F}\) via \(f\) is the \nameref{def:sheaf} \(f_{\ast}\sheaf{F}\) defined as, for every open subset \(V \subseteq Y\)
  \begin{equation*}
    f_{\ast}\sheaf{F}(V) = \sheaf{F}(f^{-1}(U))
  \end{equation*}

  This construction defines a \nameref{def:functor} \(f_{\ast}: \Sh_{X}(\mathsf{C}) \to \Sh_{Y}(\mathsf{C})\).
\end{definition}

This means that \autoref{ex:global-section-presheaf} and \autoref{ex:skyscrapper-presheaf} translate directly to \namerefs{def:sheaf}.
\begin{definition}[Global section \nameref{def:functor}] \label{def:global-section-functor}
  We denote by \(\Gamma(X, -): \Sh_{X}(\mathsf{C}) \to \mathsf{C}\) the \textbf{global section functor} described in \autoref{ex:global-section-presheaf}, but now with initial category \(\Sh_{X}(\mathsf{C})\).
\end{definition}
\begin{definition}[Skyscrapper \nameref{def:sheaf}] \label{ex:skyscrapper-sheaf}
  Let \(X\) be a topological space and \(p \in X\).
  Let \(S\) be an object of \(\mathsf{C}\).
  The \textbf{skyscrapper \nameref{def:sheaf}} supported at \(p\) \(\iota_{\ast, p} \underline{S}\) is the \nameref{def:direct-image-sheaf} of the \nameref{def:constant-sheaf} \(\underline{S}\) via the map \(\{p\} \to X\).
\end{definition}

\begin{definition}[Inverse image \nameref{def:sheaf}] \label{def:inverse-image-sheaf}
  Let \(f: X \to Y\) be a continuous map.
  Let \(\sheaf{G}\) be a \(\mathsf{C}\)-valued \nameref{def:sheaf} over \(Y\).
  The \textbf{inverse image \nameref{def:sheaf}} of \(\sheaf{G}\) via \(f\) is the \nameref{def:sheaf} \(f^{-1}\sheaf{G}\) defined as, for every open subset \(U \subseteq X\)
  \begin{equation*}
    f^{-1}\sheaf{G} = (f^{-1} \sheaf{G})^{\#}
  \end{equation*}
  where the RHS of the equation is the \nameref{def:sheafification} of the \nameref{def:inverse-image-presheaf} of \(\sheaf{G}\) via \(f\).

  This defines a \nameref{def:functor} \(f^{-1}: \PSh_{Y}(\mathsf{C}) \to \PSh_{X}(\mathsf{C})\).
\end{definition}

\begin{definition}[Constant \nameref{def:sheaf}] \label{def:constant-sheaf}
  Let \(S\) be an object of \(\mathsf{C}\).
  The \textbf{constant \nameref{def:sheaf}} \(\underline{S}\) associated to \(S\) is the \nameref{def:sheafification} of the \nameref{ex:constant-presheaf} \(\underline{S}_{\pre}\), i.e: \(\underline{S} = (\underline{S}_{\pre})^{\#}\).
\end{definition}

\begin{definition}[Stalk] \label{def:stalk-sheaf}
  Let \(X\) be a topological space and \(p \in X\). Let \(\sheaf{F}\) be a \(\mathsf{C}\)-valued \nameref{def:sheaf} over \(X\).
  The \textbf{stalk} of \(\sheaf{F}\) at \(p\) \(\sheaf{F}_{p}\) is the \nameref{def:inverse-image-sheaf} of \(\sheaf{F}\) via \(X \to \{p\}\):
  \begin{align*}
    \sheaf{F}_{p} = \colim_{U \ni p} \sheaf{F}(U)
  \end{align*}
\end{definition}


Using the adjointness we have the analogous of \autoref{prop:direct-inverse-image-adjunction-presheaves}
\begin{proposition}
  Let \(f: X \to Y\) be a continuous map.
  We have an \nameref{def:adjunction} between the \nameref{def:direct-image-sheaf} functor and the \nameref{def:inverse-image-sheaf} \(f^{-1} \dashv f_{\ast}\).
\end{proposition}
\begin{proof}
  \begin{align*}
    \Hom_{\Sh_{X}(\mathsf{C})}(f^{-1} \sheaf{G}, \sheaf{F}) &= \Hom_{\PSh_{X}(\mathsf{C})}(f^{-1} \sheaf{G}, \sheaf{F})  \\
                                                            &= \Hom_{\PSh_{X}(\mathsf{C})}(\sheaf{G}, f_{\ast}\sheaf{G}) \\
                                                            &= \Hom_{\Sh_{X}(\mathsf{C})}(\sheaf{G}, f_{\ast}\sheaf{G})
  \end{align*}
\end{proof}

Therefore \hyperref[def:adjunction]{adjunctions} \autoref{cor:constant-global-adjunction-presheaf} and \autoref{cor:stalk-constant-adjunction-presheaf} also happen when considering \namerefs{def:sheaf}.
\begin{corollary} \label{cor:constant-global-adjunction-sheaf}
  Let \(X\) be a topological space and \(p \in X\).
  Let \(\sheaf{F}\) be a \(\mathsf{C}\)-valued \namerefs{def:sheaf} over \(X\) and let \(A\) be an object in \(\mathsf{C}\).
  Then
  \begin{equation*}
    \Hom_{\Sh_{X}(\mathsf{C})}(\underline{A}, \sheaf{F}) \simeq \Hom_{\mathsf{C}}(A, \Gamma(X, \sheaf{F}))
  \end{equation*}
\end{corollary}

\begin{corollary}\label{cor:stalk-constant-adjunction-presheaf}
  Let \(X\) be a topological space and \(p \in X\).
  Let \(\sheaf{F}\) be a \(\mathsf{C}\)-valued \namerefs{def:sheaf} over \(X\) and let \(A\) be an object in \(\mathsf{C}\).
  Then
  \begin{equation*}
    \Hom_{\mathsf{C}}(\sheaf{F}_{p}, A) \simeq \Hom_{\Sh_{X}(\mathsf{C})}(\sheaf{F}, \iota_{p, \ast}\underline{A})
  \end{equation*}
\end{corollary}


For cohomology we will be interested in the case \(\mathsf{C} = \Ab\).

\begin{lemma} \label{lemma:abelian-category-from-adjunction}
  Let \(a: \mathsf{A} \to \mathsf{B}\) and \(b: \mathsf{B} \to \mathsf{A}\) be two \namerefs{def:functor}.
  Suppose that
  \begin{enumerate}
    \item \(\mathsf{A}\) and \(\mathsf{B}\) are \hyperref[def:additive-category]{additive categories}, \(a, b\) are additive functors and \(b \dashv a\).
    \item \(\mathsf{B}\) is an \nameref{def:abelian-category} and \(b\) is \hyperref[def:exact-functor]{left-exact}.
    \item \(ba \cong \id_{\mathsf{A}}\).
  \end{enumerate}
  Then \(\mathsf{A}\) is abelian.
\end{lemma}
\begin{proof}
  Since \(\mathsf{B}\) is an \nameref{def:abelian-category} all \namerefs{def:categorical-limit} and \namerefs{def:categorical-colimit} exist en \(\mathsf{B}\).

  Since \(b\) is \hyperref[def:adjunction]{left-adjoint} it is \hyperref[def:exact-functor]{right-exact} and therefore, \hyperref[def:exact-functor]{exact}.

  Let \(\varphi: B_{1} \to B_{2}\) be a morphism in \(\mathsf{B}\).
  If \(K = \ker(\varphi)\) then \(K\) is the \nameref{def:equalizer} of the diagram
  \begin{center}
    \begin{tikzcd}
      K \arrow[r] & B_{1} \arrow[r, shift left, "\varphi"] \arrow[r, shift right, "0"'] & B_{2}
    \end{tikzcd}
  \end{center}
  therefore applying \(b\) we get
  \begin{center}
    \begin{tikzcd}
      bK \arrow[r] & bB_{1} \arrow[r, shift left, "b\varphi"] \arrow[r, shift right, "0"'] & bB_{2}
    \end{tikzcd}
  \end{center}
  i.e: \(bK\) is the \nameref{def:categorical-kernel} of \(b\varphi\).

  Dually we get that \(b\coker(B_{1} \to B_{2})\) is the \nameref{def:categorical-cokernel} of \(b\varphi\).

  Therefore we have that every morphism \(b\varphi\) in \(\mathsf{A}\) admits a \nameref{def:categorical-kernel} and a \nameref{def:categorical-cokernel}.
  Moreover since \(ba \cong \id_{\mathsf{A}}\) every morphism in \(\mathsf{A}\) is of the form \(b\varphi\), therefore all \namerefs{def:categorical-kernel} and \namerefs{def:categorical-cokernel} exist in \(\mathsf{A}\).
  Indeed for every \(\psi: A_{1} \to A_{2}\) we have that
  \begin{align} \label{eq:ker-coker-lemma}
    \ker \psi = b\ker(a\psi) \qquad \coker \psi = b\coker(a\psi)
  \end{align}

  It only remains to show that \(\coim \psi = \im \psi\).
  Since \(\mathsf{A}\) has all \namerefs{def:categorical-limit} and \namerefs{def:categorical-colimit} \(\mathsf{A}\) has all finite \namerefs{def:categorical-limit} and \namerefs{def:categorical-colimit}, making \(a\) left-exact.
  This means that \(a\) transforms \namerefs{def:categorical-kernel} into \namerefs{def:categorical-kernel}.
  Therefore
  \begin{align*}
    \coim \psi &= \coker \left( \ker \psi \to A_{1} \right)      & (\text{definition}) \\
            &= b\coker \left( a(\ker \psi \to A_{1}) \right)  & \eqref{eq:ker-coker-lemma} \\
            &= b\coker \left( \ker(a\psi) \to aA_{1} \right)  & (a\text{ preserves kernels}) \\
            &= b\coim (a\psi)                                & (\text{definition}) \\
            &= b\im (a\psi)                                  & (\mathsf{B} \text{ is abelian}) \\
            &= b\ker \left( aA_{2} \to \coker(a\psi) \right)  & (\text{definition}) \\
            &= \ker \left( baA_{2} \to b\coker(a\psi) \right) & (b\text{ preserves kernels}) \\
            &= \ker \left( A_{2} \to b\coker(a\psi) \right)   & ba \cong \id_{\mathsf{A}}\\
            &= \ker \left( A_{2} \to \coker \psi \right)      & \eqref{eq:ker-coker-lemma} \\
            &= \im \psi                                     & (\text{definition})
  \end{align*}
\end{proof}

\begin{proposition}
  Let \(\varphi: \sheaf{F} \to \sheaf{G}\) be a morphism in \(\Sh(\Ab)\).
  \begin{enumerate}
    \item The \nameref{def:category} \(\Sh_{X}(\Ab)\) is an \nameref{def:abelian-category}.
    \item The \nameref{def:categorical-kernel} of \(\varphi\) is the same as the \nameref{def:categorical-kernel} of \(\varphi\) in \(\PSh(\Ab)\).
    \item \(\varphi\) is injective if and only if \(\varphi\) is injective as a map or presheaves if and only if \(\varphi\) is injective as a map of sheaves.
    \item The \nameref{def:categorical-cokernel} of \(\varphi\) is the \nameref{def:sheafification} of the \nameref{def:categorical-cokernel} of \(\varphi\) in \(\PSh(\Ab)\).
    \item \(\varphi\) is surjective if and only if \(\varphi\) is surjective as a map of sheaves.
    \item A complex \(\sheaf{F} \to \sheaf{G} \to \sheaf{H}\) is \hyperref[def:exact-complex]{exact} at \(\sheaf{G}\) if and only if for all open sets \(U\) and sections \(s \in \sheaf{G}(U)\) mapping to 0 in \(\sheaf{H}(U)\) there exists an open covering \(\{U_{i}\}_{i \in \mathcal{I}}\) such that each \(s|_{U_{i}}\) is the image of \(\sheaf{F}(U_{i}) \to \sheaf{G}(U_{i})\).
  \end{enumerate}
\end{proposition}
\begin{proof}
  Note that \(2. \implies 3.\), \(4. \implies 5\) and \(2. \land 5. \implies 6.\)

  Therefore we only need to prove \(1., 2.\) and \(3.\), but this comes directly from \autoref{lemma:abelian-category-from-adjunction} with \(\mathbf{A} = \Sh_{X}(\Ab)\), \(\mathsf{B} = \PSh_{X}(\Ab)\), \(a: \Sh_{X}(\Ab) \hookrightarrow \PSh_{X}(\Ab)\) the inclussion \nameref{def:functor} and \(b: \PSh_{X}(\Ab) \to \Sh_{X}(\Ab)\) the \nameref{def:sheafification} \nameref{def:functor}.
\end{proof}
