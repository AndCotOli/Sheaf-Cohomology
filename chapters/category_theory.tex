\chapter{Category Theory}
\begin{definition}[Category] \label{def:category}
  A \textbf{category} \(\mathsf{C}\) consists of:
  \begin{itemize}
    \item A collection of \textit{objects} \(X, Y, \ldots\)
    \item For each pair of objects \(X, Y\) a set of morphisms \(\Hom_{\mathsf{C}}(X, Y)\).
          We denote this morphisms by \(f: X \to Y\).
  \end{itemize}
  such that
  \begin{itemize}
    \item Each object has an \textit{identity morphism} \(1_{X}: X \to X\).
    \item For each pair of morphisms \(f: X \to Y\), \(g: Y \to Z\) there exists a \textit{composite morphism} \(gf: X \to Z\).
  \end{itemize}

  We impose two axioms on this data
  \begin{enumerate}
    \item For any \(f: X \to Y\) the composites \(1_{Y}f\) and \(f1_{X}\) are equal to \(f\).
    \item For any triplet of morphisms \(f: X \to Y\), \(g: Y \to W\), \(h: W \to Z\) the composites \(h(gf)\) and \((hg)f\) are equal.
          We denote this composite by \(hgf\).
  \end{enumerate}
\end{definition}
\SetPluralName{def:category}{Categories}

\begin{remark}
  Technically speaking \autoref{def:category} is the definition of a \textit{locally small category}.
  In general, a category \(\mathsf{C}\) can have an arbitrarily \textit{large} collection of morphisms between two objects.
\end{remark}

Every \nameref{def:category} has an associated opposite category
\begin{definition}[Opposite \nameref{def:category}] \label{def:opposite-category}
  Let \(\mathsf{C}\) be a \nameref{def:category}.
  The \textbf{opposite \nameref{def:category}} of \(\mathsf{C}\), \(\mathsf{C}^{\op}\) has
  \begin{itemize}
    \item The same objects as \(\mathsf{C}\).
    \item For every pair of objects \(X, Y \in \mathsf{C}^{\op}\): \(\Hom_{\mathsf{C}^{\op}}(X, Y) = \Hom_{\mathsf{C}}(Y, X)\).
  \end{itemize}
\end{definition}


\begin{definition} \label{def:mono-epi-morphism}
  A morphism \(f: x \to y\) in a \nameref{def:category} \(\mathsf{C}\) is \textbf{monomorphism} if for any parallel pair of morphisms \(h, k: w \rightrightarrows x, fh = fk\) implies \(h = k\).
  Dually, \(f\) is an \textbf{epimorphism} if for any pair of morphisms \(h, k: y \rightrightarrows z, hf = kf\) implies \(h = k\).
\end{definition}

\begin{notation}
  We denote \hyperref[def:mono-epi-morphism]{monomorphisms} by \(f: x \rightarrowtail y\) and \hyperref[def:mono-epi-morphism]{epimorphisms} by \(f: x \twoheadrightarrow y\).
\end{notation}


\begin{definition}[Isomorphism] \label{def:categorical-isomorphism}
  A morphism \(f: x \to y\) in a \nameref{def:category} \(\mathsf{C}\) is an \textbf{isomorphism} if there exists a morphism \(g: y \to x\) in \(\mathsf{C}\) such that \(fg = 1_{y}\) and \(gf = 1_{x}\).

  The objects \(x\) and \(y\) are called \textit{isomorphic} if there exists an isomorphism between them, and we denote this by \(x \cong y\).
\end{definition}

\begin{definition}[Functor] \label{def:functor}
  A \textbf{functor} \(F: \mathsf{C} \to \mathsf{D}\) between two \namerefs{def:category} consists of:
  \begin{itemize}
    \item For each object \(c\) in \(\mathsf{C}\) an object \(Fc \in \mathsf{D}\).
    \item For each morphism \(f: c \to c'\) in \(\mathsf{C}\) a morphism \(Ff: Fc \to Fc'\) in \(\mathsf{D}\).
  \end{itemize}
  satisfying the following axioms
  \begin{enumerate}
    \item For each element \(c\) in \(\mathsf{C}\), \(F 1_{c} = 1_{Fc}\).
    \item For any composable pair of morphisms \(f, g\) in \(\mathsf{C}\), \(Fg \cdot Ff = F(g \cdot f)\).
  \end{enumerate}
\end{definition}

\begin{notation}
  A \nameref{def:functor} of the form \(\mathsf{C} \to \mathsf{D}\) is usually called a \textit{covariant functor} from \(\mathsf{C}\) to \(\mathsf{D}\) while a functor of the form \(\mathsf{C}^{\op} \to \mathsf{D}\) is called a \textit{contravariant} functor from \(\mathsf{C}\) to \(\mathsf{D}\).
\end{notation}


\begin{example}[Posetal category of open sets] \label{ex:open-set-category}
  The \nameref{def:functor} \(\open: \Top \to \mathsf{Poset}\) sends a topological space \(X\) to its poset of open sets.

  Therefore for every topological space with have an associated \textit{posetal category of open sets} \(\open(X)\).
\end{example}


\begin{definition}[Natural transformation] \label{def:natural-transformation}
  Let \(\mathsf{C}, \mathsf{D}\) be two \namerefs{def:category} and \(F, G: \mathsf{C} \rightrightarrows \mathsf{D}\) be two \namerefs{def:functor}.
  A \textbf{natural transformation} \(\alpha: F \Rightarrow G\) consists of:
  \begin{itemize}
    \item For each object \(c \in \mathsf{C}\) an arrow \(\alpha_{c}: Fc \to Gc\) in \(\mathsf{D}\) such that, for any morphism \(f: c \to c'\) in \(\mathsf{C}\) the following diagram commutes in \(\mathsf{D}\)
          \begin{center}
            \begin{tikzcd}
              Fc \arrow[r, "\alpha_{c}"] \arrow[d, "Ff"'] & Gc \arrow[d, "Gf"] \\
              Fc' \arrow[r, "\alpha_{c'}"']               & Gc'
            \end{tikzcd}
          \end{center}
  \end{itemize}

  We denote this by
  \begin{center}
    \begin{tikzcd}
      \mathsf{C} \arrow[r, bend left=50, "F"{name=U, above}] \arrow[r, bend right=50, "G"{name=D, below}] & \mathsf{D} \arrow[shorten <= 5pt, shorten >= 5pt, Rightarrow, from=U, to=D, "\alpha"]
    \end{tikzcd}
  \end{center}
\end{definition}

\begin{definition} \label{def:natural-isomorphism}
  A \nameref{def:natural-transformation} \(\alpha: F \Rightarrow G\) is a \textbf{natural isomorphism} if every component \(\alpha_{f}\) is an \nameref{def:categorical-isomorphism}.

  In this case, we say that \(F\) and \(G\) are \textbf{naturally isomorphic} and we denote it by \(\alpha: F \cong G\).
\end{definition}


\begin{definition} \label{def:functor-types}
  A \nameref{def:functor} \(F: \mathsf{C} \to \mathsf{D}\) is
  \begin{itemize}
    \item \textbf{full} if for all pair of objects \(x, y \in \mathsf{C}\) the map \(\Hom_{\mathsf{C}}(x, y) \to \Hom_{\mathsf{D}}(Fx, Fy)\) is surjective.
    \item \textbf{faithful} if for all pair of objects \(x, y \in \mathsf{C}\) the map \(\Hom_{\mathsf{C}}(x, y) \to \Hom_{\mathsf{D}}(Fx, Fy)\) is injective.
    \item \textbf{essentially surjective on objects} if for every object \(d \in \mathsf{D}\) there exists \(c \in \mathsf{C}\) such that \(d \cong Fc\).
  \end{itemize}

  If \(F\) is both full and faithfull we will say it is \textbf{fully faithful}.
\end{definition}

\begin{definition}[Concrete category] \label{def:concrete-category}
  A \nameref{def:category} \(\mathsf{C}\) is a \textbf{concrete \nameref{def:category}} if there exists a \nameref{def:functor} \(F: \mathsf{C} \to \Set\) that is \hyperref[def:functor-types]{faithful}.
\end{definition}


\begin{definition}[Diagram] \label{def:diagram}
  A \textbf{diagram} in a \nameref{def:category} \(\mathsf{C}\) with shape \(\mathsf{J}\) is a \nameref{def:functor} \(F: \mathsf{C} \to \mathsf{J}\).
\end{definition}

\begin{definition}[Initial and terminal objects] \label{def:initial-terminal-objects}
  Let \(\mathsf{C}\) be a \nameref{def:category}.
  An object \(i \in \mathsf{C}\) is \textbf{initial} if for every object \(c \in \mathsf{C}\) there exists a unique morphism \(i \to c\).
  Dually, an object \(t \in \mathsf{C}\) is \textbf{terminal} if for every object \(c \in \mathsf{C}\) there exists a unique morphism \(c \to t\).
\end{definition}

\begin{definition} \label{def:functors-represented-by}
  Let \(c\) be an object in a \nameref{def:category} \(\mathsf{C}\).
  The \textbf{\nameref{def:functor} represented} by \(c\) is the \nameref{def:functor} \(\Hom_{\mathsf{C}}(c, -) \to \Set\) defined as
  \begin{center}
    \begin{tikzcd}
      x \arrow[d, "f"{name=L, left}] & \Hom_{\mathsf{C}}(c, x) \arrow[d, "f_{\ast}"{name=R}]  \\
      y & \Hom_{\mathsf{C}}(c, y) \\
      \arrow[shorten <= 10pt, shorten >= 10pt, mapsto, from=L, to=R]
    \end{tikzcd}
  \end{center}
\end{definition}

\begin{remark}
  If we consider the \nameref{def:opposite-category} \(\mathsf{C}^{\op}\) in \autoref{def:functors-represented-by} for each \(c \in \mathsf{C}^{\op}\) we get a \nameref{def:functor} \(\Hom_{\mathsf{C}^{\op}}(c, -) \to \Set\) defined as
  \begin{center}
    \begin{tikzcd}
      x \arrow[d, "f"{name=L, left}] & \Hom_{\mathsf{C}^{\op}}(c, x) \arrow[d, "f_{\ast}"{name=R}]  \\
      y & \Hom_{\mathsf{C}^{\op}}(c, y) \\
      \arrow[shorten <= 10pt, shorten >= 10pt, mapsto, from=L, to=R]
    \end{tikzcd}
  \end{center}

  Therefore since \(\Hom_{\mathsf{C}^{\op}}(c, x) = \Hom_{\mathsf{C}}(x, c)\) we get a functor \(\Hom_{\mathsf{C}}(-, c) \to \Set\)
  \begin{center}
    \begin{tikzcd}
      x \arrow[d, "f"{name=L, left}] & \Hom_{\mathsf{C}}(x, c) \\
      y & \Hom_{\mathsf{C}}(y, c) \arrow[u, "f^{\ast}"{name=R, right}]
      \arrow[shorten <= 10pt, shorten >= 10pt, mapsto, from=L, to=R]
    \end{tikzcd}
  \end{center}
\end{remark}


\begin{definition}
  A \nameref{def:functor} \(F: \mathsf{C} \to \Set\) is \textbf{representable} if there exists \(c \in \mathsf{C}\) such that \(F\) is \hyperref[def:natural-isomorphism]{naturally isomorphic} to the \hyperref[def:functors-represented-by]{represented functor} associated to \(c\).
\end{definition}
