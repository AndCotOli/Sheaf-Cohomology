\chapter{Cohomology of sheaves}

In what follows, let \(\mathbf{k}\) be a fixed commutative ring.

We want to define the cohomology of \(X\) with coefficients in a \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} as the \hyperref[def:right-derived-functor]{right derived functor} of the \hyperref[def:global-sections-sheaf]{global sections functor}
\begin{align*}
  \Gamma(X, -)\colon \Mod(\underline{\mathbf{k}}_{X}) \to \Mod(\mathbf{k})
\end{align*}
To do this, we need to check the following:
\begin{enumerate}
  \item \(\Mod(\underline{\mathbf{k}_{X}})\) and \(\Mod(\mathbf{k})\) are \hyperref[def:abelian-category]{abelian categories}.
  \item The \hyperref[def:functor]{functor} \(\Gamma(X, -)\) is \hyperref[def:exact-functor]{left exact}.
  \item \(\Mod(\underline{\mathbf{k}}_{X})\) has \hyperref[def:enough-injectives-projectives]{enough injectives}.
\end{enumerate}

Note that \(\Mod(\mathbf{k})\) is an \hyperref[def:abelian-category]{abelian category}.
Therefore by \autoref{thm:sheaf-category-is-abelian} \(\Mod(\underline{\mathbf{k}}_{X})\) is also \hyperref[def:abelian-category]{abelian}.

Recall that by \autoref{cor:global-constant-adjunction-sheaf} \(\Gamma(X, -)\) is \hyperref[def:adjoint-functors]{right adjoint}.
Therefore by \autoref{ex:adjoint-functors-and-exactness} \(\Gamma(X, -)\) is \hyperref[def:exact-functor]{left exact}.

It only remains to show that \(\Mod(\underline{\mathbf{k}}_{X})\) has \hyperref[def:enough-injectives-projectives]{enough injectives}.
\begin{lemma}\label{lemma:mod-ox-has-enough-injectives}
  The \hyperref[def:category]{category} \(\Mod(\underline{\mathbf{k}}_{X})\) of \hyperref[def:sheaf]{sheaves of \(\mathbf{k}\)-modules} over \(X\) has \hyperref[def:enough-injectives-projectives]{enough injectives}.
\end{lemma}
\begin{proof}
  We will use the following claim
  \begin{quote}
    The \hyperref[def:category]{category} \(\Mod(R)\) of modules over a ring \(R\) has enough injectives.
  \end{quote}
  For a proof, see \cite[Corollary VIII.6.4]{aluffiAlgebraChapter02009}.
  
  Let \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{X})\) and consider its \hyperref[def:stalk]{stalk} \(\sheaf{F}_{p} \in \Mod(\mathbf{k})\) at a point \(p \in X\).

  Since \(\Mod(\mathbf{k})\) has \hyperref[def:enough-injectives-projectives]{enough injectives} we can choose an object \(I_{p}\) and a \hyperref[def:mono-epi-morphism]{monomorphism}
  \begin{equation*}
    i_{p}\colon \sheaf{F}_{p} \monoarrow I_{p}
  \end{equation*}

  Consider the \hyperref[def:sheaf]{sheaf}
  \begin{equation*}
    \sheaf{I} \coloneq \bigoplus_{p \in X} \operatorname{sk}_{p} \underline{I_{p}}
  \end{equation*}
  where \(\operatorname{sk}_{p}\underline{I_{p}}\) is the \hyperref[def:skyscraper-sheaf]{skyscraper sheaf} supported at the \hyperref[def:constant-sheaf]{constant sheaf} \(\underline{I_{p}}\).
  More explicitly, for every open set \(U \subseteq X\) the \hyperref[def:sheaf]{sheaf} \(\sheaf{I}\) is defined as
  \begin{equation*}
    \sheaf{I}(U) = \bigoplus_{p \in U} I_{p}
  \end{equation*}

  Recall that by \autoref{cor:skyscraper-stalk-adjunction-sheaf} the \hyperref[def:skyscraper-sheaf]{skyscraper functor} is \hyperref[def:adjoint-functors]{left-adjoint} to the \hyperref[def:stalk]{stalk functor}, which is \hyperref[def:exact-functor]{exact}.
  Therefore by \autoref{prop:adjoints-and-injectives} \(\sheaf{I}\) is an \hyperref[def:injective-projective-object]{injective object} of \(\Mod(\underline{\mathbf{k}}_{X})\).
\end{proof}

This means that we can \hyperref[def:right-derived-functor]{derive} all \hyperref[def:exact-functor]{right-exact functors} from the \hyperref[def:sheaf]{category of sheaves of \(\mathbf{k}\)-modules}.

Having seen this, we now can finally define sheaf cohomology.
\begin{definition}[Sheaf Cohomology]\label{def:sheaf-cohomology}
  Let \(\sheaf{F}\) be a \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} over a topological space \(X\).
  The \(i\)-th \textbf{cohomology} of \(X\) with coefficients in \(\sheaf{F}\) is the \(i\)-th \hyperref[def:right-derived-functor]{right derived functor} of the \hyperref[def:global-sections-sheaf]{global sections functor}:
  \begin{equation*}
    \cohomology{i}(X; \sheaf{F}) \coloneq R^{i}\Gamma(X, \sheaf{F})
  \end{equation*}
\end{definition}



Therefore, given a \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{X})\) to compute its \hyperref[def:sheaf-cohomology]{cohomology} we \textit{only} need to construct an \hyperref[def:injective-projective-object]{injective} (or \hyperref[def:F-acyclic-object]{\(\Gamma(X, -)\) acyclic} by \autoref{thm:right-derived-functor-using-acyclics}) \hyperref[def:resolution]{resolution} of \(\sheaf{F}\)
\begin{center}
  \begin{tikzcd}
    0 \arrow[r] & \sheaf{F}^{(0)} \arrow[r] & \sheaf{F}^{(1)} \arrow[r] & \cdots
  \end{tikzcd}
\end{center}
and compute the \hyperref[def:cohomology]{cohomology} of the \(\mathbf{k}\)-modules
\begin{center}
  \begin{tikzcd}
    0 \arrow[r] & \Gamma(X, \sheaf{F}^{(0)}) \arrow[r] & \Gamma(X, \sheaf{F}^{(1)}) \arrow[r] & \cdots
  \end{tikzcd}
\end{center}

Since \hyperref[def:resolution]{injective resolutions} are usually hard to find (note how \autoref{lemma:mod-ox-has-enough-injectives} doesn't give an \textit{easy} construction) we will mostly work with \hyperref[def:F-acyclic-object]{\(\Gamma(X, -)\)-acyclic resolutions}.

\begin{lemma}\label{lemma:cohomology-open-subsets}
  Let \(X\) be a topological space and \(U \subseteq X\) an open subset.
  \begin{enumerate}
    \item If \(\sheaf{I}\) is an \hyperref[def:injective-projective-object]{injective} \hyperref[def:sheaf]{sheaf} over \(X\) then \(\sheaf{I}|_{U}\) is \hyperref[def:injective-projective-object]{injective} over \(U\).
    \item For any \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{X})\) we have \(\cohomology{i}(U; \sheaf{F}) = \cohomology{i}(U; \sheaf{F}|_{U})\).
  \end{enumerate}
\end{lemma}
\begin{proof}
  Let \(j\colon U \hookrightarrow X\) be the open immersion.
  \hyperref[thm:open-immersion-adjunction-sheaf]{There is an adjunction \(j_{!} \dashv j^{-1}\)}.
  Furthermore, \hyperref[cor:extension-by-zero-sheaf-exact]{\(j_{!}\) is exact}.
  Therefore 1 follows from \autoref{prop:adjoints-and-injectives}.

  To see 2, by \hyperref[def:sheaf-cohomology]{definition} \(\cohomology{i}(U; \sheaf{F}) = \cohomology{i}(\Gamma(U, \sheaf{I}^{\bullet}))\) where \(\sheaf{F} \to \sheaf{I}^{\bullet}\) is an \hyperref[def:resolution]{injective resolution}.
  By 1 \(\sheaf{F}|_{U} \to \sheaf{I}^{\bullet}|_{U}\) is an \hyperref[def:resolution]{injective resolution} of \(\sheaf{F}|_{U}\) in \(\Mod(\underline{\mathbf{k}}_{U})\).
  Therefore
  \begin{equation*}
    \cohomology{i}(U; \sheaf{F}|_{U}) = \cohomology{i}(\Gamma(U, \sheaf{I}^{\bullet}|_{U}))  = \cohomology{i}(U; \sheaf{F}) \qedhere
  \end{equation*}
\end{proof}

\begin{lemma}\label{lemma:cohomology-closed-subsets}
  Let \(X\) be a topological space and \(Z \subseteq X\) a closed subset.
  \begin{enumerate}
    \item If \(\sheaf{I}\) is an \hyperref[def:injective-projective-object]{injective} \hyperref[def:sheaf]{sheaf} over \(X\) then \(\sheaf{I}_{Z}\) is \hyperref[def:injective-projective-object]{injective} over \(Z\).
    \item For any \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{X})\) we have \(\cohomology{i}(X; \sheaf{F}_{Z}) = \cohomology{i}(Z; \sheaf{F}|_{Z})\).
  \end{enumerate}
\end{lemma}
\begin{proof}
  Analogous to \autoref{lemma:cohomology-open-subsets} using the closed immersion \(i\colon Z \hookrightarrow X\).
\end{proof}

\begin{lemma}\label{lemma:cohomology-compact-subsets}
  Let \(X\) be a topological space and \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{X})\).
  If \(K \subseteq X\) is compact and relatively Hausdorff then
  \begin{equation*}
    \varinjlim_{\substack{U \supset K \\ U \in \Open(X)}}  \cohomology{i}(U; \sheaf{F}) \cong \cohomology{i}(K; \sheaf{F})
  \end{equation*}
\end{lemma}
\begin{proof}
  Suppose first that \(i = 0\).
  In this case we want to show that there is an \hyperref[def:categorical-isomorphism]{isomorphism} induced by the restriction
  \begin{equation*}
    \varinjlim_{\substack{U \supset K \\ U \in \Open(X)}} \Gamma(U, \sheaf{F}) \isoarrow \Gamma(K, \sheaf{F})
  \end{equation*}
  Let \(s \in \Gamma(K, \sheaf{F}|_{K})\).
  There exists a finite family of open subsets \(\{U_{i}\}_{i=1}^{n}\) covering \(K\) and sections \(s_{i} \in \Gamma(U_{i}, \sheaf{F})\) such that \(s_{i}|_{K \cap U_{i}} = s|_{K \cap U_{i}}\).
  Furthermore we can find another family of open sets \(\{V_{i}\}_{i=1}^{n}\) covering \(K\) such that \(K \cap \operatorname{ad}(V_{i}) \subset U_{i}\).
  We want to glue together the sections \(s_{i}\) on a neighbourhood of \(K\).

  By induction over \(n\) we can assume \(n = 2\).
  If \(K_{i} = K \cap \operatorname{ad}(V_{i})\) then \(s_{1}|_{K_{1} \cap K_{2}} = s_{2}|_{K_{1} \cap K_{2}}\).
  Let \(W\) be an open neighbourhood of \(K\) such that \(s_{1}|_{K} = s_{2}|_{K}\) and let \(W_{i}\) be an open cover subset of \(U_{i}\) such that \(K_{i} \setminus W \subset W_{i}\) and \(W_{1} \cap W_{2} = \varnothing\).
  Set \(U_{i}' = W_{i} \cup W\).
  Then \(s_{1}|_{U_{1}' \cap U_{2}'} = s_{2}|_{U_{1}' \cap U_{2}'}\) and this defines \(t \in \Gamma(U_{1}' \cup U_{2}', \sheaf{F})\) with \(t|_{K} = s\) finishing the base case.
  
  Now, consider two \hyperref[def:functor]{functors} \(F, G\colon \Mod(\underline{\mathbf{k}}_{X}) \to \Mod(\mathbf{k})\)
  \begin{equation*}
    F(\sheaf{F}) \coloneq \varinjlim_{\substack{U \supset K \\ U \in \Open(X)}} \Gamma(U, \sheaf{F}) \qquad G(\sheaf{F}) \coloneq \Gamma(K, \sheaf{F})
  \end{equation*}
  From the first step we have \(F \cong G\).
  Therefore for all \(i \geq 0\) we have an \hyperref[def:categorical-isomorphism]{isomorphism}
  \begin{equation*}
    \operatorname{R}^{i}F(\sheaf{F}) \cong \operatorname{R}^{i}G(\sheaf{F})
  \end{equation*}
  To compute these functors, choose an \hyperref[def:resolution]{injective resolution} \(\sheaf{F} \to \sheaf{I}^{\bullet}\) of \(\sheaf{F}\).
  On the one hand
  \begin{equation*}
    \operatorname{R}^{i}G(\sheaf{F}) = \cohomology{i}\Gamma(K; I^{\bullet}) = \cohomology{i}(K; \sheaf{F})
  \end{equation*}
  On the other
  \begin{align*}
    \operatorname{R}^{i}F(\sheaf{F}) = \cohomology{i} \left( \varinjlim_{\substack{U \supset K \\ U \in \Open(X)}} \Gamma(U, \sheaf{I}^{\bullet}) \right) \cong \varinjlim_{\substack{U \supset K \\ U \in \Open(X)}} \cohomology{i} \left( \Gamma(U, \sheaf{I^{\bullet}}) \right) = \varinjlim_{\substack{U \supset K \\ U \in \Open(X)}} \cohomology{i} \left( U; \sheaf{F} \right)
  \end{align*}
  Therefore
  \begin{equation*}
    \varinjlim_{\substack{U \supset K \\ U \in \Open(X)}} \cohomology{i} \left( U; \sheaf{F} \right) \cong \operatorname{R}^{i}F(\sheaf{F}) \cong \operatorname{R}^{i}G(\sheaf{F}) = \cohomology{i}(K; \sheaf{F}) \qedhere
  \end{equation*}
\end{proof}

\begin{theorem}[Open Mayer-Vietoris]\label{thm:mayer-vietoris-open}
  Let \(X\) be a topological space.
  Suppose that \(X = U \cup V\) for some open sets \(U, V\).
  Then for every \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{X})\) there exists a \hyperref[def:exact-sequence]{long exact sequence} in \hyperref[def:cohomology]{cohomology}:
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \cohomology{0}(X; \sheaf{F}) \arrow[r] & \cohomology{0}(U; \sheaf{F}) \oplus \cohomology{0}(V; \sheaf{F}) \arrow[r] \arrow[d, phantom, ""{coordinate, name=X}] & \cohomology{0}(U \cap V; \sheaf{F}) \arrow[dll, rounded corners, to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (X) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      & \cohomology{1}(X; \sheaf{F}) \arrow[r] & \cohomology{1}(U; \sheaf{F}) \oplus \cohomology{1}(V; \sheaf{F}) \arrow[r] \arrow[d, phantom, ""{coordinate, name=Y}] &  \cohomology{1}(U \cap V; \sheaf{F}) \arrow[dll, rounded corners, to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (Y) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      & \cohomology{2}(X; \sheaf{F}) \arrow[r] & \cohomology{2}(U; \sheaf{F}) \oplus \cohomology{2}(V; \sheaf{F}) \arrow[r] & \cdots
    \end{tikzcd}
  \end{center}
\end{theorem}
\begin{proof}
  Consider the sequence of \hyperref[def:sheaf]{sheaves of \(\mathbf{k}\)-modules}
  \begin{center}
    \begin{tikzcd}[column sep=huge]
      0 \arrow[r] & \sheaf{F} \arrow[r, "{s \mapsto (s|_{U}, s|_{V})}"] & \sheaf{F}|_{U} \oplus \sheaf{F}|_{V} \arrow[r, "{(s, t)} \mapsto s|_{U \cap V} - t|_{U \cap V}"] & \sheaf{F}|_{U \cap V} \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  Let \(p \in X = U \cup V\).
  If \(p \in U\) but \(p \notin V\) the induced sequence on the \hyperref[def:stalk]{stalks} is
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \sheaf{F}_{p} \arrow[r] & \sheaf{F}_{p} \oplus 0 \arrow[r] & 0 \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  which is \hyperref[def:exact-sequence]{exact}.
  Analogously if \(p \in V\) but \(p \notin U\) we also get an \hyperref[def:exact-sequence]{exact sequence}.

  Now, if \(p \in U \cap V\) we have
  \begin{center}
    \begin{tikzcd}[column sep=huge]
      0 \arrow[r] & \sheaf{F}_{p} \arrow[r, "{s \mapsto (s_{x}, s_{x})}"] & \sheaf{F}_{p} \oplus \sheaf{F}_{p} \arrow[r, "{(s, t) \mapsto s - t}"] & \sheaf{F}_{p} \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  which is also \hyperref[def:exact-sequence]{exact}.

  Therefore
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \sheaf{F} \arrow[r] & \sheaf{F}|_{U} \oplus \sheaf{F}|_{V} \arrow[r] & \sheaf{F}|_{U \cap V} \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  is \hyperref[def:exact-sequence]{exact} in \(\Mod(\underline{\mathbf{k}}_{X})\).
  Applying \autoref{thm:LES-derived-functor} with \(\Gamma(X, -)\) and using \autoref{lemma:cohomology-open-subsets} we get the result.
\end{proof}

\begin{theorem}[Closed Mayer-Vietoris]\label{thm:mayer-vietoris-closed}
  Let \(X\) be a topological space.
  Suppose that \(X = Z \cup W\) for some closed sets \(Z, W\).
  Then for every \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{X})\) there exists a \hyperref[def:exact-sequence]{long exact sequence} in \hyperref[def:cohomology]{cohomology}:
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \cohomology{0}(X; \sheaf{F}) \arrow[r] & \cohomology{0}(X; \sheaf{F}_{Z}) \oplus \cohomology{0}(X; \sheaf{F}_{W}) \arrow[r] \arrow[d, phantom, ""{coordinate, name=X}] & \cohomology{0}(X; \sheaf{F}_{Z \cap W}) \arrow[dll, rounded corners, to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (X) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      & \cohomology{1}(X; \sheaf{F}_{Z}) \arrow[r] & \cohomology{1}(X; \sheaf{F}_{Z}) \oplus \cohomology{1}(X; \sheaf{F}_{W}) \arrow[r] \arrow[d, phantom, ""{coordinate, name=Y}] &  \cohomology{1}(X; \sheaf{F}_{Z \cap W}) \arrow[dll, rounded corners, to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (Y) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      & \cohomology{2}(X; \sheaf{F}) \arrow[r] & \cohomology{2}(X; \sheaf{F}_{U}) \oplus \cohomology{2}(X; \sheaf{F}_{V}) \arrow[r] & \cdots
    \end{tikzcd}
  \end{center}
\end{theorem}
\begin{proof}
  Analogous to the \hyperref[thm:mayer-vietoris-open]{open version} with
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \sheaf{F} \arrow[r] & \sheaf{F}_{Z} \oplus \sheaf{F}_{W} \arrow[r] & \sheaf{F}_{Z \cap W} \arrow[r] & 0 \qedhere
    \end{tikzcd}
  \end{center}
\end{proof}

\section{Acyclic resolutions}

\begin{definition}[Flasque sheaf]\label{def:flasque-sheaf}
  A \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} \(\sheaf{F}\) is \textbf{flasque} (or \textbf{flabby}) if for any inclussion of open subsets \(V \subseteq U\) the restriction map \(\res_{U, V}\colon \sheaf{F}(U) \to \sheaf{F}(V)\) is surjective.
\end{definition}

\begin{proposition}\label{prop:flasque-are-acyclic}
  \hyperref[def:flasque-sheaf]{Flasque sheaves} are \hyperref[def:acyclic-object]{\(\Gamma(X, -)\)-acyclic}.
\end{proposition}
\begin{proof}
  Let \(\sheaf{F}\) be a \hyperref[def:flasque-sheaf]{flasque sheaf}.
  Since \(\Mod(\underline{\mathbf{k}}_{X})\) has \hyperref[def:enough-injectives-projectives]{enough injectives} we have a \hyperref[def:exact-sequence]{short exact sequence}
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \sheaf{F} \arrow[r] & \sheaf{I} \arrow[r] & \sheaf{G} \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  where \(\sheaf{I}\) is an \hyperref[def:injective-projective-object]{injective object}.
  We will first show that, for any open subset \(U \subseteq X\), the map \(\sheaf{I}(U) \to \sheaf{G}(U)\) is surjective.
  
  Let \(s \in \sheaf{G}(U)\).
  Let \(V, W \subseteq U\) be two open subsets and \(t_{V} \in \sheaf{I}(V), t_{W} \in \sheaf{I}(W)\) be two sections lifting \(s\) to \(\sheaf{I}\).
  Consider
  \begin{equation*}
    t_{V}|_{V \cap U} - t_{W}|_{V \cap W} \in \sheaf{F}(V \cap W)
  \end{equation*}
  Since \(\sheaf{F}\) is \hyperref[def:flasque-sheaf]{flasque} there exists a section \(x_{V} \in \sheaf{F}(V)\) such that \(x_{V}|_{V \cap W} = t_{V} - t_{W}\).
  Therefore if \(t'_{V} = t_{V} - x_{V}\) we have \(t'_{V}|_{V \cap W} = t_{W}|_{V \cap W}\), i.e: there exists a section \(t \in \sheaf{I}(V \cup W)\) such that \(t|_{V} = t'|_{V}\) and \(t'|_{W} = t|_{W}\).
  Clearly \(t\) is sent to \(s|_{V \cap W}\).

  Consider now a pair \((W, t)\) which is maximal (for the obvious order relation), where \(W\) is an open subset of \(U\) and \(t\) is a section of \(\sheaf{I}\) lifting \(s\) to \(W\).
  Since \(s\) lifts locally on \(\sheaf{I}\) and using the previous result we see that \(W = U\).

  But then using the \hyperref[thm:LES-in-cohomology]{long exact sequence in cohomology} of our sequence and the fact that for \(i > 0 \quad \cohomology{i}(X, \sheaf{I}) = 0\) imply that \(\cohomology{1}(X, \sheaf{F}) = 0\) we get
  \begin{equation*}
    \forall k - 1 \geq 1 \quad \cohomology{k}(X, \sheaf{F}) = \cohomology{k-1}(X, \sheaf{G})
  \end{equation*}
  Furthermore surjectivity of the maps \(\sheaf{I}(U) \epiarrow \sheaf{G}(U)\) implies that \(\sheaf{G}\) is \hyperref[def:flasque-sheaf]{flasque}.

  It follows by induction that
  \begin{equation*}
    \forall k > 0 \quad \cohomology{k}(X, \sheaf{F}) = 0 \qedhere
  \end{equation*}
\end{proof}

Using \hyperref[def:flasque-sheaf]{flasque sheaves} we can construct a canonical (and functorial) resolution of a \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} to compute its \hyperref[def:sheaf-cohomology]{cohomology}.
\begin{construction}[Godement resolution]\label{def:godement-resolution}
  Let \(\sheaf{F}\) be a \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules}.
  The \textbf{Godement resolution} of \(\sheaf{F}\) is constructed by considering the inclussion of \(\sheaf{F}\) into the \textit{Godement sheaf} \(\sheaf{F}_{\operatorname{God}}\)
  \begin{equation*}
    U \mapsto \bigoplus_{p \in U} \sheaf{F}_{p} \eqcolon \sheaf{F}_{\operatorname{God}}(U)
  \end{equation*}
  Note that \(\sheaf{F}_{\operatorname{God}}\) is \hyperref[def:flasque-sheaf]{flasque}.

  We then inject the quotient \(\quot{\sheaf{F}_{\operatorname{God}}}{\sheaf{F}}\) into \(\left( \quot{\sheaf{F}_{\operatorname{God}}}{\sheaf{F}} \right)_{\operatorname{God}}\) and so on.
\end{construction}

\begin{example}
  Let's try to compute \hyperref[def:sheaf-cohomology]{sheaf cohomology} using the \hyperref[def:godement-resolution]{Godement resolution}.

  Let \(X = \{p, q\}\) with the \textit{Sierpiński} topology \(\mathcal{T} = \{\varnothing, \{p\}, X\}\).
  Consider the \hyperref[def:constant-sheaf]{constant sheaf} supported at the integers \(\sheaf{F} = \underline{\mathbb{Z}}\).

  To compute the \hyperref[def:sheaf-cohomology]{cohomology} of \(\sheaf{F}\), we are going to construct the complex of global sections:
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \Gamma(X, \sheaf{F}_{\operatorname{God}}) \arrow[r, "\d^{0}"] & \Gamma\left(X, \left(\quot{\sheaf{F}_{\operatorname{God}}}{\sheaf{F}} \right)_{\operatorname{God}}\right) \arrow[r, "\d^{1}"] & \cdots
    \end{tikzcd}
  \end{center}

  Note that the \hyperref[def:stalk-ox-mod]{stalks} of \(\sheaf{F}\) are:
  \begin{equation*}
    \sheaf{F}_{p} = \sheaf{F}(\{p\}) = \mathbb{Z}, \qquad \sheaf{F}_{q} = \sheaf{F}(X) = \mathbb{Z}.
  \end{equation*}

  Consider the Godement sheaf associated to \(\sheaf{F}\):
  \begin{equation*}
    \sheaf{F}_{\operatorname{God}}(U) = \bigoplus_{x \in U} \sheaf{F}_{x}.
  \end{equation*}
  We have that:
  \begin{equation*}
    \Gamma(X, \sheaf{F}_{\operatorname{God}}) = \sheaf{F}_{\operatorname{God}}(X) = \sheaf{F}_{p} \oplus \sheaf{F}_{q} = \mathbb{Z} \oplus \mathbb{Z}.
  \end{equation*}
  This means that we have a map \(\d^{-1}\) as follows:
  \begin{equation*}
    \d^{-1}: \underbrace{\Gamma(X, \sheaf{F})}_{\mathbb{Z}} \to \underbrace{\Gamma(X, \sheaf{F}_{\operatorname{God}})}_{\mathbb{Z} \oplus \mathbb{Z}}, \quad \d^{-1}(k) = (k, k).
  \end{equation*}
  Let \(\sheaf{Q}^1 = \quot{\sheaf{F}_{\operatorname{God}}}{\sheaf{F}}\) be the \hyperref[def:categorical-cokernel]{cokernel} sheaf.
  If we compute its stalks, we have that:
  \begin{align*}
    (\sheaf{Q}^1)_{p} &= \quot{\sheaf{F}_{\operatorname{God}, p}}{\sheaf{F}_{p}} = \quot{\mathbb{Z}}{\mathbb{Z}} = 0, \\
    (\sheaf{Q}^1)_{q} &= \quot{\sheaf{F}_{\operatorname{God}, q}}{\sheaf{F}_{q}} = \quot{\mathbb{Z} \oplus \mathbb{Z}}{\langle(k, k)\rangle} \cong \mathbb{Z}.
  \end{align*}
  Therefore, the global sections of the Godement sheaf of \(\sheaf{Q}^{1}\) are:
  \begin{equation*}
    \Gamma\left(X, (\sheaf{Q}^1)_{\operatorname{God}}\right) = (\sheaf{Q}^1)_{p} \oplus (\sheaf{Q}^1)_{q} = 0 \oplus \mathbb{Z} = \mathbb{Z}.
  \end{equation*}
  This produces a differential map:
  \begin{equation*}
    \d^{0}: \underbrace{\Gamma(X, \sheaf{F}_{\operatorname{God}})}_{\mathbb{Z} \oplus \mathbb{Z}} \to \underbrace{\Gamma\left(X, (\sheaf{Q}^1)_{\operatorname{God}}\right)}_{\mathbb{Z}}, \quad \d^{0}(m, n) = m - n.
  \end{equation*}
  Let \(\sheaf{Q}^2\) be the \hyperref[def:categorical-cokernel]{cokernel} of the map \(\sheaf{F}_{\operatorname{God}} \to (\sheaf{Q}^1)_{\operatorname{God}}\).
  If we compute its \hyperref[def:stalk-ox-mod]{stalks}, we have:
  \begin{align*}
    (\sheaf{Q}^2)_{p} &= \quot{0}{0} = 0, \\
    (\sheaf{Q}^2)_{q} &= \quot{\mathbb{Z}}{\mathbb{Z}} = 0.
  \end{align*}
  Therefore, we have that:
  \begin{equation*}
    \Gamma \left( X, (\sheaf{Q}^2)_{\operatorname{God}} \right) = 0 \oplus 0 = 0.
  \end{equation*}
  This means that our resolution of global sections is:
  \begin{center}
    \begin{tikzcd}[column sep=huge]
      0 \arrow[r] & \mathbb{Z} \oplus \mathbb{Z} \arrow[r, "d^{0}", "{(m, n) \mapsto m - n}"'] & \mathbb{Z} \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  Therefore, the \hyperref[def:sheaf-cohomology]{cohomology groups} of \(X\) with coefficients in \(\underline{\mathbb{Z}}\) are:
  \begin{align*}
    \cohomology{0}(X, \underline{\mathbb{Z}}) &= \ker(d^{0}) = \{(m, n) \in \mathbb{Z} \oplus \mathbb{Z} \mid m - n = 0\} \cong \mathbb{Z}, \\
    \cohomology{1}(X, \underline{\mathbb{Z}}) &= \operatorname{coker}(d^{0}) = \quot{\mathbb{Z}}{\operatorname{im}(d^{0})} = \quot{\mathbb{Z}}{\mathbb{Z}} = 0, \\
    \cohomology{i}(X, \underline{\mathbb{Z}}) &= 0 \quad \forall i > 1.
  \end{align*}
\end{example}
