\chapter{Čech Cohomology}

\begin{definition}[\v{C}ech complex]\label{def:cech-complex}
  Let \((X, \sheaf{O}_{X})\) be a \hyperref[def:ringed-space]{ringed space} and let \(\sheaf{F}\) be a \hyperref[def:sheaf-of-modules]{sheaf of \(\sheaf{O}_{X}\)}-modules.

  Consider an open covering \(\mathcal{U} = \{U_{i}\}_{i \in \mathbb{N}}\) and denote
  \begin{equation*}
    U_{i_{0}, \ldots, i_{p}} \coloneq U_{i_{0}} \cap \cdots \cap U_{i_{p}}
  \end{equation*}
  For all \(p \geq 0\) set
  \begin{equation*}
    \check{\mathcal{C}}^{p}(\mathcal{U}, \sheaf{F}) = \prod_{(i_{0}, \ldots, i_{p})} \sheaf{F}(U_{i_{0}, \ldots, i_{p}})
  \end{equation*}
  and for all \(s \in \check{C}^{p}(\mathcal{U}, \sheaf{F})\) denote by \(s_{i_{0}, \ldots, i_{p}}\) its value in \(\sheaf{F}(U_{i_{0}, \ldots, i_{p}})\).

  Define the \textit{differential maps} \(d^{p}: \check{\mathcal{C}}^{p}(\mathcal{U}, \sheaf{F}) \to \check{\mathcal{C}}^{p+1}(\mathcal{U}, \sheaf{F})\) as
  \begin{equation*}
    d^{p}(s)_{i_{0}, \ldots, i_{p}} \coloneq \sum_{j=0}^{p+1} (-1)^{j} s_{i_{0}, \ldots, \hat{\imath_{0}}, \ldots, i_{p+1}}|_{U_{i_{0}, \ldots, i_{p+1}}}
  \end{equation*}

  A straightforward calculation shows that \(d^{p+1} \circ d^{p} = 0\), making \(\check{C}^{\bullet}(\mathcal{U}, \sheaf{F})\) a \hyperref[def:complex]{complex}.
  We call this the \textbf{\v{C}ech complex} associated to \(\sheaf{F}\) and \(\mathcal{U}\).

  Note that we can define a \textit{natural} map \(\varepsilon: \sheaf{F} \to \check{\mathcal{C}}(\mathcal{U}, \sheaf{F})\) by sending \(s \in \sheaf{F}(U)\) to \(\varepsilon(s)_{i} \coloneq s|_{U \cap U_{i}}\).
\end{definition}

\begin{proposition}\label{prop:cech-complex-is-resolution}
  Let \((X, \sheaf{O}_{X})\) be a \hyperref[def:ringed-space]{ringed space} and let \(\sheaf{F}\) be a \hyperref[def:sheaf-of-modules]{sheaf of \(\sheaf{O}_{X}\)}-modules.

  Then the \hyperref[def:cech-complex]{\v{C}ech complex} associated to \(\sheaf{F}\) and \(\mathcal{U}\) is a \hyperref[def:resolution]{resolution} of \(\sheaf{F}\).

\begin{definition}[\v{C}ech cohomology]\label{def:cech-cohomology}
  Let \((X, \sheaf{O}_{X})\) be a \hyperref[def:ringed-space]{ringed space} and let \(\sheaf{F}\) be a \hyperref[def:sheaf-of-modules]{sheaf of \(\sheaf{O}_{X}\)}-modules.
  Let \(\mathcal{U}\) be an open covering of \(X\).

  The \textbf{\v{C}ech \hyperref[def:cohomology]{cohomology}} associated to \(\sheaf{F}\) and the covering \(\mathcal{U}\) is
  \begin{equation*}
    \check{\operatorname{H}}^{\bullet}(\mathcal{U}, \sheaf{F}) \coloneq \cohomology{\bullet}(\check{\mathcal{C}}^{\bullet}(\mathcal{U}, \sheaf{F}))
  \end{equation*}
\end{definition}

\begin{theorem}
  Let \((X, \sheaf{O}_{X})\) be a \hyperref[def:ringed-space]{ringed space}, where \(X\) is Hausdorff and paracompact.
  Let \(\sheaf{F}\) be a \hyperref[def:sheaf-of-modules]{sheaf of \(\sheaf{O}_{X}\)-modules}.

  Then for all open coverings \(\mathcal{U}\)
  \begin{equation*}
    \check{\operatorname{H}}^{\bullet}(\mathcal{U}, \sheaf{F}) \cong \cohomology{\bullet}(X, \sheaf{F})
  \end{equation*}
\end{theorem}
\begin{proof}
  \cite[Theorem 5.10.1]{godementTopologieAlgebriqueTheorie1998}
\end{proof}
