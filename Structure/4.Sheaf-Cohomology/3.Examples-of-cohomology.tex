\chapter{Some examples of cohomology}

We focus our attention now to the computation of the cohomology of some topological spaces.
Our main tools will be the \hyperref[thm:mayer-vietoris-open]{Mayer-Vietoris sequence} and the \hyperref[thm:homotopy-invariance-cohomology]{homotopy invariance of cohomology}.

We will not go into much detail into the explicit description of the partitions used of our target space, relying more on geometric intuition.

The main reference for this chapter is \cite{kashiwaraSheavesManifolds1990}.

\begin{example}[\(\mathbb{S}^{1}\)]\label{ex:cohomology-S1}
  Let \(\sheaf{F} \in \Mod( \underline{\mathbf{k}}_{S} )\) be a \hyperref[def:sheaf]{sheaf of \(\mathbf{k}\)-modules} over \(S\) and \(X \coloneq \mathbb{S}^{1} \times S\).
  
  Consider the open cover of \(\mathbb{S}^{1} = H^{+} \cup H^{-}\) where \(H^{+}\) is the upper hemisphere and \(H^{-}\) the lower one, both extended slightly past the equator.
  \begin{center}
    \begin{tikzpicture}[scale=1]
      % Coordinates
      \coordinate (A) at (0.00, 0.00);
      \coordinate (B) at (-3.00, 0.00);
      \coordinate (C) at (-0.02, 2.71);
      \coordinate (D) at (0.00, -2.50);
      \coordinate (E) at (3.00, 0.00);
      \coordinate (F) at (0.08, -0.05);

      % Drawing
      \draw[thick] (A) circle (2.00);
      \node[above] at (F) {$\mathbb{S}^{1}$};

      \draw[blue] (2.41, -0.65) arc (-15.0:195.0:2.50);
      \node[above, text=blue] at (C) {$H^{+}$};
      
      \draw[red] (-2.17, 0.58) arc (165.0:375.0:2.25);
      \node[below, text=red] at (D) {$H^{-}$};
      
      \draw[purple] (-2.66, 0.71) arc (165.0:195.0:2.75);
      \node[left, text=purple] at (B) {$W_{1}$};
      
      \draw[purple] (2.66, -0.71) arc (-15.0:15.0:2.75);
      \node[right, text=purple] at (E) {$W_{2}$};
    \end{tikzpicture}
  \end{center}
  
  Let \(U = H^{+} \times S\) and \(V = H^{-} \times S\).
  Since \(H^{+}\) and \(H^{-}\) are homeomorphic to open intervals, which are contractible, both \(U\) and \(V\) are \hyperref[def:homotopy-over]{contractible over \(S\)}.
  Furthermore \(U \cap V = \left( W_{1} \times S \right) \coprod \left( W_{2} \times S \right)\) consists of two disjoint pieces, each of these \hyperref[def:homotopy-over]{contractible over \(S\)}.

  Therefore since \hyperref[cor:cohomology-and-contractible]{cohomology of contractible spaces is concentrated in degree 0} we only have
  \begin{align*}
    \operatorname{R}^{0} p_{U, \ast}\sheaf{F}|_{U} &\cong \sheaf{F} \\
    \operatorname{R}^{0}p_{V, \ast}\sheaf{F}|_{V} &\cong \sheaf{F} \\
    \operatorname{R}^{0}p_{U \cap V, \ast} \sheaf{F}|_{U \cap V} &\cong \sheaf{F} \oplus \sheaf{F}
  \end{align*}
  Writing the \hyperref[thm:mayer-vietoris-open]{Mayer-Vietoris sequence} since the higher terms vanish we end up with
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \operatorname{R}^{0} p_{X, \ast} \sheaf{F}|_{X} \arrow[r] & \sheaf{F} \oplus \sheaf{F} \arrow[r, "\alpha"] & \sheaf{F} \oplus \sheaf{F} \arrow[r, "\delta"] & \operatorname{R}^{1} p_{X, \ast} \sheaf{F}|_{X} \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  Therefore directly we get \(\operatorname{R}^{i}p_{X, \ast}\sheaf{F}_{X} = 0\) for all \(i \geq 2\).

  Note that \(\alpha\colon \sheaf{F} \oplus \sheaf{F} \to \sheaf{F} \oplus \sheaf{F}\) maps a pair of sections \((s_{U}, s_{V})\) into their difference on the overlap.
  Since the section \(s_{U}\) is defined globally on \(U\) its restriction to \(W_{1}\) and to \(W_{2}\) is identical and analogously for \(s_{V}\).
  Therefore we have
  \begin{equation*}
    \alpha(s_{U}, s_{V}) = (s_{U} - s_{V}, s_{U} - s_{V})
  \end{equation*}
  and thus \(\ker \alpha = \{(s, s) \mid s \in \sheaf{F}\}\) is the diagonal which means that
  \begin{equation*}
    \operatorname{R}^{0}p_{X, \ast} \sheaf{F}|_{X} \cong \sheaf{F}
  \end{equation*}

  On the other hand by \hyperref[def:exact-sequence]{exactness} \(\delta\) is surjective, meaning that
  \begin{equation*}
    \operatorname{R}^{1}p_{X, \ast}\sheaf{F}|_{X} \cong \coker \alpha \cong \quot{\sheaf{F} \oplus \sheaf{F}}{\operatorname{Im} \alpha} = \quot{\sheaf{F} \oplus \sheaf{F}}{\langle(s, s)\rangle} \cong \sheaf{F}
  \end{equation*}

  Therefore
  \begin{equation*}
    \operatorname{R}^{i}p_{\mathbb{S}^{1} \times S, \ast}\sheaf{F}|_{\mathbb{S}^{1} \times S} = \begin{cases}
      \sheaf{F} & i = 0 \\
      \sheaf{F} & i = 1 \\
      0 & i \geq 2
    \end{cases}
  \end{equation*}
  In particular taking \(S = \{\ast\}\) since \(\sheaf{F} \cong \underline{M}\) for some \(M \in \Mod(\mathbf{k})\) we get
  \begin{equation*}
    \cohomology{i}(\mathbb{S}^{1}; \underline{M}) = \begin{cases}
      M & i = 0 \\
      M & i = 1 \\
      0 & i \geq 2
    \end{cases}
  \end{equation*}
\end{example}

\begin{example}[\(\mathbb{S}^{n}\)]\label{ex:cohomology-Sn}
  Let \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{S})\).
  We are going to see that cohomology of \(X = \mathbb{S}^{n} \times S\) over \(S\) by induction over \(n\) is
  \begin{equation*}
    \operatorname{R}^{i}p_{X, \ast}\sheaf{F}|_{X} = \begin{cases}
      \sheaf{F} & i = 0 \\
      \sheaf{F} & i = n \\
      0 & i \notin \{0, n\}
    \end{cases}
  \end{equation*}
  We have already computed the base case \(n = 1\) in \autoref{ex:cohomology-S1}, so we can suppose \(n \geq 2\).
  
  Consider the open cover \(\mathbb{S}^{n} = H^{+} \cup H^{-}\) where \(H^{+}\) is the open upper hemisphere and \(H^{-}\) is the lower open hemisphere, both extending slightly past the equator.

  We have that \(U \coloneq H^{+} \times S\) and \(V \coloneq H^{-} \times S\) are \hyperref[def:homotopy-over]{contractible over \(S\)}.
  Furthermore since \(H^{+} \cap H^{-}\) is a thickened equatorial band it can be \textit{squeezed} into the equator itself, which it is a \((n-1)\)-sphere.
  This means that \(H^{+} \cap H^{-}\) \hyperref[def:deformation-retract]{strong-deformation retracts} onto \(\mathbb{S}^{n-1}\) and thus, \(U \cap V\) \hyperref[def:deformation-retract]{strong-deformation retracts} onto \(\mathbb{S}^{n-1} \times S\).

  Since \(\mathbb{S}^{n-1}\) is connected the first terms of the \hyperref[thm:mayer-vietoris-open]{Mayer-Vietoris sequence} are
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \operatorname{R}^{0} p_{X, \ast} \sheaf{F}|_{X} \arrow[r] & \sheaf{F} \oplus \sheaf{F} \arrow[r, "\alpha"] \arrow[d, phantom, ""{coordinate, name=X}] & \sheaf{F} \arrow[dll, rounded corners, "\delta", to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (X) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      & \operatorname{R}^{1} p_{X, \ast} \sheaf{F}|_{X} \arrow[r] & 0 \oplus 0 \arrow[r] & \cdots
    \end{tikzcd}
  \end{center}
  In this case we have \(\alpha(s_{U}, s_{V}) = s_{U} - s_{V}\) and thus \(\operatorname{R}^{0}p_{X, \ast}\sheaf{F}|_{X} \cong \sheaf{F}\).
  Furthermore since \(\alpha\) is clearly surjective and \(\delta\) is surjective because the \hyperref[def:exact-sequence]{sequence is exact} we have \(\operatorname{R}^{1}p_{X, \ast}\sheaf{F}|_{X} \cong 0\).

  If we focus now on the higher terms of the \hyperref[thm:mayer-vietoris-open]{Mayer-Vietoris sequence} we have, for \(i \geq 1\)
  \begin{center}
    \begin{tikzcd}
      \cdots \arrow[r] & \operatorname{R}^{i}p_{U, \ast} \oplus \operatorname{R}^{i}p_{V, \ast} \arrow[r] \arrow[d, phantom, ""{coordinate, name=X}] & \operatorname{R}^{i}p_{U \cap V, \ast} \arrow[dll, rounded corners, "\delta", to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (X) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      \operatorname{R}^{i+1}p_{X, \ast} \arrow[r] & \operatorname{R}^{i+1}p_{U, \ast} \oplus \operatorname{R}^{i+1}p_{V, \ast} \arrow[r] & \cdots
    \end{tikzcd}
  \end{center}
  Since \(U\) and \(V\) are \hyperref[def:homotopy-over]{contractible over \(S\)} these terms are zero by \autoref{cor:cohomology-and-contractible} and thus we get an \hyperref[def:categorical-isomorphism]{isomorphism}
  \begin{equation*}
    \operatorname{R}^{i}p_{U \cap V, \ast} \isoarrow \operatorname{R}^{i+1}p_{X, \ast}
  \end{equation*}
  Substituting \(U \cap V\) we get
  \begin{equation*}
    \operatorname{R}^{i+1}p_{X, \ast} \sheaf{F}|_{\mathbb{S}^{n} \times S} \cong \operatorname{R}^{i}p_{X, \ast} \sheaf{F}|_{\mathbb{S}^{n-1} \times S} \cong \begin{cases}
      \sheaf{F} & i \in \{0, n-1\} \\
      \sheaf{F} & i \notin \{0, n-1\}
    \end{cases}
  \end{equation*}
  
  Therefore
  \begin{equation*}
    \operatorname{R}^{i}p_{\mathbb{S}^{n} \times S, \ast}\sheaf{F}|_{\mathbb{S}^{n} \times S} = \begin{cases}
      \sheaf{F} & i = 0 \\
      \sheaf{F} & i = n \\
      0 & i \notin \{0, n\}
    \end{cases}
  \end{equation*}
  In particular taking \(S = \{\ast\}\) since \(\sheaf{F} \cong \underline{M}\) for some \(M \in \Mod(\mathbf{k})\) we get
  \begin{equation*}
    \cohomology{i}(\mathbb{S}^{n}; \underline{M}) = \begin{cases}
      M & i = 0 \\
      M & i = n \\
      0 & i \notin \{0, n\}
    \end{cases}
  \end{equation*}
\end{example}

\begin{example}[\(\mathbb{R}^{n} \setminus \{0\}\)]
  We are going to see first that \(X \coloneq \left( \mathbb{R}^{n} \setminus \{0\} \right) \times S\) is a \hyperref[def:deformation-retract]{strong deformation retract} of \(A \coloneq \mathbb{S}^{n-1} \times S\).
  To do this, consider the inclussion \(i\colon A \hookrightarrow X\) and the retraction \(r\colon X \to A, r(v, s) \coloneq \left( \frac{v}{\lVert v\rVert}, s \right)\).
  As the \hyperref[def:homotopy-over]{homotopy} we can take
  \begin{align*}
    H\colon& X \times [0, 1] \to A \\
    &(v, s, t) \mapsto \left( (1 - t)v + t \frac{v}{\lVert v \rVert}, s \right)
  \end{align*}
  Therefore by \autoref{cor:cohomology-and-retraction} for all \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{S})\) we have and \(i \geq 0\)
  \begin{equation*}
    \operatorname{R}^{i}p_{X, \ast}\sheaf{F}|_{X} \cong \operatorname{R}^{i}p_{A, \ast}\sheaf{F}|_{X}
  \end{equation*}
  In particular if \(n \geq 2\) we have
  \begin{equation*}
    \operatorname{R}^{i}p_{\left(\mathbb{R}^{n} \setminus \{0\}\right) \times S, \ast}\sheaf{F}|_{\left( \mathbb{R}^{n} \setminus \{0\} \right) \times S} \cong \begin{cases}
      \sheaf{F} & i = 0 \\
      \sheaf{F} & i = n-1 \\
      0 & i \notin \{0, n-1\}
    \end{cases}
  \end{equation*}
  The case \(n = 1\) is different since \(\mathbb{R} \setminus \{0\}\) has two connected components.
  We have a \hyperref[def:deformation-retract]{deformation retract} \(\mathbb{R} \setminus \{0\} \to \mathbb{S}^{0} \cong \{\pm 1\}\).
  Therefore
  \begin{equation*}
    \operatorname{R}^{i}p_{\left( \mathbb{R} \setminus \{0\} \right) \times S, \ast}\sheaf{F}|_{\left( \mathbb{R} \setminus \{0\} \right) \times S} \cong \begin{cases}
      \sheaf{F} \oplus \sheaf{F} & i = 0 \\
      0 & i \neq 0
    \end{cases}
  \end{equation*}

  In particular taking \(S = \{\ast\}\) since \(\sheaf{F} \cong \underline{M}\) for some \(M \in \Mod(\mathbf{k})\) we get
  \begin{align*}
     \cohomology{i}(\mathbb{R} \setminus \{0\}; \underline{M}) &= \begin{cases}
      M \oplus M & i = 0 \\
      0 & i \neq 0
     \end{cases} \\
    \cohomology{i}(\mathbb{R}^{n} \setminus \{0\}; \underline{M}) &= \begin{cases}
      M & i = 0 \\
      M & i = n-1 \\
      0 & i \notin \{0, n-1\}
    \end{cases} \qquad \forall n > 1
  \end{align*}
\end{example}

\begin{example}[\(\mathbb{RP}^{2}\)]\label{ex:cohomology-RP2}
  Let \(\sheaf{F} \in \Mod(\underline{\mathbf{k}}_{S})\) and \(X \coloneq \mathbb{RP}^{2} \times S\).

  Consider an open cover \(\mathbb{RP}^{2} = U_{0} \cup V_{0}\) where \(U_{0}\) is an open disk around a point, \(V_{0}\) is an open neighbourhood of the \textit{equator} \(\mathbb{RP}^{1} \subset \mathbb{RP}^{2}\) (which is homeomorphic to an open Möbius band).
  In this case \(U_{0} \cap V_{0}\) is homeomorphic to an open annulus.
  \begin{center}
    \begin{tikzpicture}[
      scale=1.5,
      % Style for the identified boundary of RP^2
      antipodal/.style={
        thick,
        decoration={
          markings,
          mark=at position 0.25 with {\arrow{>}},
          mark=at position 0.75 with {\arrow{>}}
        },
        postaction={decorate}
      }
      ]
      % Define the radii for our sets
      \def\R{2}     % Full disk radius (the "equator" RP^1)
      \def\Ru{1.75}  % Outer radius of U_0
      \def\Rv{1.25}  % Inner radius of V_0

      % Draw V_0 (Neighborhood of the equator)
      % Shaded blue. It extends from radius \Rv to \R.
      \fill[blue!15] (0,0) circle (\R);
      \fill[white] (0,0) circle (\Rv);

      % Draw U_0 (Open disk around the pole)
      % Shaded red, slightly transparent to show the intersection
      \fill[red!20, opacity=0.8] (0,0) circle (\Ru);

      % Dashed lines to indicate these are open sets
      \draw[dashed, red!80!black, thick] (0,0) circle (\Ru);
      \draw[dashed, blue!80!black, thick] (0,0) circle (\Rv);

      % Draw the equator (RP^1) with antipodal identification arrows
      \draw[antipodal] (0,0) circle (\R);

      % --- Labels ---
      
      % Label for U_0
      \node[red!80!black, font=\large] at (0, 0.7) {$U_0$};

      % Label for V_0
      \node[blue!80!black, font=\large] at (0, 2.7) {$V_0$};

      % Label for the intersection
      \node[purple!80!black, font=\large] at (0, 1.9) {$U_0 \cap V_0$};
    \end{tikzpicture}
  \end{center}
  
  If we let \(U \coloneq U_{0} \times S\) and \(V \coloneq V_{0} \times S\) then \(X = U \cup V\) and we can use \hyperref[thm:mayer-vietoris-open]{Mayer-Vietoris} to compute the cohomology.

  On the one hand since \(U_{0}\) is contractible \(U\) is \hyperref[def:homotopy-over]{contractible over \(S\)} and thus by \autoref{cor:cohomology-and-contractible}
  \begin{equation*}
    \operatorname{R}^{i}p_{U, \ast} \sheaf{F}|_{U} \cong \begin{cases}
      \sheaf{F} & i = 0 \\
      0 & i > 0
    \end{cases}
  \end{equation*}

  On the other since \(V_{0}\) and \(U_{0} \cap V_{0}\) strongly deformation retract into their inner circle \(\mathbb{S}^{1}\) both \(V\) and \(U \cap V\) strongly deformation retract over \(S\) into \(\mathbb{S}^{1} \times S\) and thus by \autoref{cor:cohomology-and-retraction} and \autoref{ex:cohomology-S1}
  \begin{equation*}
    \operatorname{R}^{i}p_{V, \ast} \sheaf{F}_{V} = \operatorname{R}^{i}p_{U \cap V} \cong \begin{cases}
      \sheaf{F} & i \in \{0, 1\} \\
      0 & i > 1
    \end{cases}
  \end{equation*}

  Therefore the associated \hyperref[thm:mayer-vietoris-open]{Mayer-Vietoris sequence} is
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \operatorname{R}^{0}p_{X, \ast}\sheaf{F}|_{X} \arrow[r] \arrow[d, phantom, ""{coordinate, name=X}] & \sheaf{F} \oplus \sheaf{F} \arrow[r, "\alpha"] & \sheaf{F} \arrow[dll, rounded corners, "\delta_{0}", to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (X) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      & \operatorname{R}^{1}p_{X, \ast}\sheaf{F}|_{X} \arrow[r] \arrow[d, phantom, ""{coordinate, name=Y}] & 0 \oplus \sheaf{F} \arrow[r, "\beta"] & \sheaf{F} \arrow[dll, rounded corners, "\delta_{1}", to path={
        -- ([xshift=2ex]\tikztostart.east)
        |- (Y) [near end]\tikztonodes
        -| ([xshift=-2ex]\tikztotarget.west)
        -- (\tikztotarget)}] \\
      & \operatorname{R}^{2}p_{X, \ast}\sheaf{F}|_{X} \arrow[r] & 0
    \end{tikzcd}
  \end{center}

  The map \(\alpha\) is given by \((s_{1}, s_{2}) \mapsto s_{1} - s_{2}\) and thus its \hyperref[def:categorical-kernel]{kernel} is the diagonal making
  \begin{equation*}
    \operatorname{R}^{0}p_{X, \ast}\sheaf{F}|_{X} \cong \sheaf{F}
  \end{equation*}
  Furthermore since \(\alpha\) is surjective we get \(\operatorname{Im} \alpha = \sheaf{F}\) and thus \(\delta_{0} = 0\).

  The map \(\beta\) is induced by the inclussion \(U \cap V \hookrightarrow V\).
  Therefore since the annulus \(U_{0} \cap V_{0}\) wraps twice around the inner circle of \(V_{0}\) we have that \(\beta\) is induced by multiplication by two.
  Therefore by exactness
  \begin{equation*}
    \operatorname{R}^{i}p_{\mathbb{RP}^{2} \times S, \ast}\sheaf{F}|_{\mathbb{RP}^{2} \times S} \cong \begin{cases}
      \sheaf{F} & i = 0 \\
      \ker \left(\sheaf{F} \xrightarrow{\cdot 2} \sheaf{F} \right) & i = 1 \\
      \coker \left( \sheaf{F} \xrightarrow{\cdot 2} \sheaf{F} \right) = \quot{\sheaf{F}}{2 \sheaf{F}} & i = 2 \\
      0 & i > 2
    \end{cases}
  \end{equation*}
  where \(2\sheaf{F}\) denotes the image of \(\sheaf{F}\) by the induced map of multiplication by two.

  In particular taking \(S = \{\ast\}\) since \(\sheaf{F} \cong \underline{M}\) for some \(M \in \Mod(\mathbf{k})\) we get
  \begin{equation*}
    \cohomology{i}(\mathbb{RP}^{2}; \underline{M}) \cong \begin{cases}
      M & i = 0 \\
      \ker \left( M \xrightarrow{\cdot 2} M \right) & i = 1 \\
      \quot{M}{2 M} & i = 2 \\
      0 & i > 2
    \end{cases}
  \end{equation*}
\end{example}
