\section{Complexes and cohomology}

Throughout this chapter we will be working over a fixed \hyperref[def:abelian-category]{abelian category} \(\category{A}\).

\begin{definition}[Complex] \label{def:complex}
  A sequence of objects and morphisms
  \begin{center}
    \begin{tikzcd}
      \cdots \arrow[r, "d_{i-2}"] & C_{i-1} \arrow[r, "d_{i-1}"] & C_{i} \arrow[r, "d_{i}"] & C_{i+1} \arrow[r, "d_{i+1}"] & \cdots
    \end{tikzcd}
  \end{center}
  in \(\category{A}\) is a \textbf{chain complex} if for every \(i \in \mathbb{N} \quad d_{i-1} \circ d_{i} = 0\).
  We denote a chain complex as \((C_{\bullet}, d_{\bullet})\) or simply as \(C_{\bullet}\).

  By reversing the order, we say that a sequence
  \begin{center}
    \begin{tikzcd}
      \cdots \arrow[r, "d^{i+2}"] & C^{i+1} \arrow[r, "d^{i+1}"] & C^{i} \arrow[r, "d^{i}"] & C^{i-1} \arrow[r, "d^{i-1}"] & \cdots
    \end{tikzcd}
  \end{center}
  is a \textbf{cochain complex} if for every \(i \in \mathbb{N} \quad d^{i} \circ d^{i+1} = 0\).
  We denote a cochain complex as \((C^{\bullet}, d^{\bullet})\) or simply as \(C^{\bullet}\).
\end{definition}

\begin{remark}
  In purelly algebraic terms, chains are cochains are \textit{equivalent}.
  We will adopt the cochain notation, but all proofs and definitions translate to the chain notation.
\end{remark}


\begin{definition}[Exact complex] \label{def:exact-complex}
  A \hyperref[def:complex]{complex}
  \begin{center}
    \begin{tikzcd}
      \cdots \arrow[r, "d^{i+2}"] & C^{i+1} \arrow[r, "d^{i+1}"] & C^{i} \arrow[r, "d^{i}"] & C^{i-1} \arrow[r, "d^{i-1}"] & \cdots
    \end{tikzcd}
  \end{center}
  in \(\category{A}\) is \textbf{exact at \(C^{i}\)} (or simply at position \(i\)) if
  \begin{equation*}
    \coker d^{i+1} \circ \ker d^{i} = 0
  \end{equation*}

  A complex is \textbf{exact} if it is exact at every position.
\end{definition}

\begin{definition}[Short exact sequence] \label{def:short-exact-sequence}
  An \hyperref[def:exact-complex]{exact complex} of the form
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & X \arrow[r] & Y \arrow[r] & Z \arrow[r] & 0
    \end{tikzcd}
  \end{center}
  is called a \textbf{short exact sequence}.

  \hyperref[def:exact-complex]{Exact complexes} of the form \(0 \to X \to Y \to Z\) (resp. \(X \to Y \to  Z \to 0\)) are called left (resp. right) short exact sequences.
\end{definition}


\begin{remark}
  For any morphism \(f: X \to Y\) in an \(\category{A}\) we get \hyperref[def:short-exact-sequence]{short exact sequences}
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & \ker f \arrow[r] & X \arrow[r] & \im f \arrow[r] & 0 \\
      0 \arrow[r] & \im f \arrow[r] & Y \arrow[r] & \coker f \arrow[r] & 0
    \end{tikzcd}
  \end{center}

  Therefore by \autoref{lemma:partial-sequence-additive} the sequence \(0 \to X \xrightarrow{f} Y\) (resp. \(X \xrightarrow{f} Y \to 0\)) is \hyperref[def:exact-sequence]{exact} if and only if \(f\) is a \hyperref[def:mono-epi-morphism]{monomorphism} (resp. \hyperref[def:mono-epi-morphism]{epimorphism}).
\end{remark}

\begin{definition}[Exact functor] \label{def:exact-functor}
  Let \(\category{A}, \category{B}\) be two \hyperref[def:abelian-category]{abelian categories}.

  An \hyperref[def:additive-functor]{additive functor} \(F: \category{A} \to \category{B}\) is called left (resp. right) \textbf{exact} if for any \hyperref[def:short-exact-sequence]{short exact sequence} in \(\category{A}\)
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & X \arrow[r] & Y \arrow[r] & Z
    \end{tikzcd}
  \end{center}
  (resp. \(X \to Y \to Z \to 0\) ) the squence
  \begin{center}
    \begin{tikzcd}
      0 \arrow[r] & F(X) \arrow[r] & F(Y) \arrow[r] & F(Z)
    \end{tikzcd}
  \end{center}
  (resp. \(F(X) \to F(Y) \to F(Z) \to 0\)) is \hyperref[def:short-exact-sequence]{exact}.

  If \(F\) is both left and right exact, then \(F\) is called \textbf{exact}.
\end{definition}

\begin{example}
  The \hyperref[def:functor]{functors} \(\Hom_{\category{A}}(X, -)\) and \(\Hom_{\category{A}}(-, X)\) are both \hyperref[def:exact-functor]{left-exact functors} from \(\category{A}\) to \(\Ab\).
\end{example}

The following is \cite[Lemma 5.2.2]{schapiraIntroductionCategoriesSheaves}.
\begin{lemma}
  Let \(F: \category{A} \to \category{B}\) be an \hyperref[def:additive-functor]{additive functor}.
  \begin{enumerate}
    \item The following are equivalent
          \begin{enumerate}
            \item \(F\) is \hyperref[def:exact-functor]{left exact}.
            \item \(F\) commutes with \hyperref[def:categorical-kernel]{kernels}: for any morphism \(f: X \to Y\), \(F(\ker f) \isoarrow \ker F(f)\).
            \item \(F\) commutes with \hyperref[def:categorical-limit]{finite limits}.
            \item For any \hyperref[def:short-exact-sequence]{short exact sequence} \(0 \to X \to Y \to Z \to 0\) in \(\category{A}\) the sequence \(0 \to F(X) \to F(Y) \to F(Z)\) is \hyperref[def:short-exact-sequence]{exact} in \(\category{B}\).
          \end{enumerate}

    \item  The following are equivalent
          \begin{enumerate}
            \item \(F\) is \hyperref[def:exact-functor]{exact}.
            \item For any \hyperref[def:exact-complex]{exact sequence} \(X \to Y \to Z\) in \(\category{A}\) the sequence \(F(X) \to F(Y) \to F(Z)\) is \hyperref[def:exact-complex]{exact} in \(\category{B}\).
            \item \(F\) commutes with \hyperref[def:categorical-colimit]{finite limits} and \hyperref[def:categorical-colimit]{finite colimits}.
          \end{enumerate}
  \end{enumerate}
\end{lemma}

\begin{definition} \label{def:injective-projective-objects}~
  \begin{itemize}
    \item An object \(P \in \category{A}\) is \textbf{projective} if all the diagrams of the form
          \begin{center}
            \begin{tikzcd}
              & X \arrow[d, two heads] \\
              P \arrow[r] & Z
            \end{tikzcd}
          \end{center}
          where \(X \epiarrow X\) is an \hyperref[def:mono-epi-morphism]{epimorphism} admit an extension
           \begin{center}
             \begin{tikzcd}
               & X \arrow[d, two heads] \\
               P \arrow[r] \arrow[ur, dotted] & Z
            \end{tikzcd}
          \end{center}

    \item An object \(I \in \category{A}\) is \textbf{injective} if all the diagrams of the form
          \begin{center}
            \begin{tikzcd}
              X \arrow[r] \arrow[d, tail] & I \\
              Z
            \end{tikzcd}
          \end{center}
          where \(X \monoarrow Z\) is a \hyperref[def:mono-epi-morphism]{monomorphism} admit an extension
           \begin{center}
            \begin{tikzcd}
              X \arrow[r] \arrow[d, tail] & I \\
              Z \arrow[ur, dotted]
            \end{tikzcd}
          \end{center}
  \end{itemize}
\end{definition}

\begin{proposition}
  An object \(I \in \category{A}\) is \hyperref[def:injective-projective-objects]{injective} if and only if the functor \(\Hom_{\category{A}}(-, I)\) is \hyperref[def:exact-functor]{exact}.

  An object \(P \in \category{A}\) is \hyperref[def:injective-projective-objects]{projective} if and only if the functor \(\Hom_{\category{A}}(P, -)\) is \hyperref[def:exact-functor]{exact}.
\end{proposition}


\begin{definition} \label{def:enough-projective-injectives}~
  \begin{itemize}
    \item \(\category{A}\) has \textbf{enough projectives} if for every object \(A \in \category{A}\) there exists a \hyperref[def:injective-projective-objects]{projective object} \(P \in \category{A}\) and an \hyperref[def:mono-epi-morphism]{epimorphism} \(P \epiarrow A\).
    \item \(\category{A}\) has \textbf{enough injectives} if for every object \(A \in \category{A}\) there exists a \hyperref[def:injective-projective-objects]{injective object} \(I \in \category{A}\) and an \hyperref[def:mono-epi-morphism]{monomorphism} \(A \monoarrow I\).
  \end{itemize}
\end{definition}
