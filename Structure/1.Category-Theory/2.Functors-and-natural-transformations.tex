\chapter{Functors and natural transformations}\label{ch:functors-natural-transformations}

Having established the "static" structure of categories, we now turn to the "dynamic" relationships between them.
In the study of sheaves, objects are rarely considered in isolation; instead, we are concerned with how data is moved between categories—such as the way a continuous map \(f\colon X \to Y\) induces a relationship between the collections of open sets \(\Open(X)\) and \(\Open(Y)\).
Functors provide the formal mechanism for these assignments, while natural transformations allow us to compare different functors.

This language is not merely a convenience; it is a necessity for defining the Derived Functors that appear later in this work.
Specifically, we will see that a sheaf is itself a functor, and the process of "sheafification" is a natural transformation that mediates between the world of presheaves and sheaves.

\section{Functors}

\begin{definition}[Functor]\label{def:functor}
  Let \(\category{C}, \category{D}\) be two \hyperref[def:category]{categories}.
  A (covariant) \textbf{functor} \(F: \category{C} \to \category{D}\) consists of:
  \begin{itemize}
    \item for each object \(X \in \category{C}\), an object \(F(X) \in \category{D}\).
    \item for each morphism \(f: X \to Y\) in \(\category{C}\), a morphism \(Ff: F(X) \to F(Y)\) in \(\category{D}\),
  \end{itemize}
  such that:
  \begin{itemize}
    \item \(F(\id_X) = \id_{F(X)}\) for every \(X \in \category{C}\).
    \item \(F(g \circ f) = Fg \circ Ff\) for every composable pair of morphisms \(f, g\) in \(\category{C}\).
  \end{itemize}

  A functor is \textbf{contravariant} if it reverses the order of composition: \(F(g \circ f) = Ff \circ Fg\). This is equivalent to a covariant functor \(F: \opcategory{C} \to \category{D}\).

  Diagramatically, a covariant functor is represent by
  \begin{center}
    \begin{tikzcd}
      \llap{\(F\colon\)} \category{C} \arrow[r] & \category{D} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & F(X) \arrow[d, "Ff" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & F(Y)
    \end{tikzcd}
  \end{center}
  And a contravariant functor by
  \begin{center}
    \begin{tikzcd}
      \llap{\(F\colon\)} \category{C} \arrow[r] & \category{D} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & F(X)  \arrow[from=source, to=target, mapsto] \\
      Y  & F(Y) \arrow[u, "Ff"' ""{name=target, left}]
    \end{tikzcd}
  \end{center}
\end{definition}

\begin{remark}
  We will disfavour the covariant/contravariant distinction.
  Therefore all \hyperref[def:functor]{functors} will be covariant (unless otherwise specified) and contravariant \hyperref[def:functor]{functors} will be covariant \hyperref[def:functor]{functors} from the \hyperref[def:opposite-category]{opposite category}.
\end{remark}

\begin{example}~
  \begin{itemize}
    \item There is a (covariant) \hyperref[def:functor]{endofunctor} \(P\colon \Set \to \Set\) that sends a set \(A\) to its power set \(PA = \{ A' \subseteq A\}\) and a map \(f\colon A \to B\) to the direct image \(f_{\ast}\colon PA \to PB, f_{\ast}(A') \coloneq f(A')\).

          We also get a contravariant version \(P\colon \Set^{\op} \to \Set\) that sends a map \(f\colon A \to B\) to the inverse image \(f^{-1}\colon PB \to PA, f^{-1}(B') \coloneq f^{-1}(B)\).
    \item There is a \hyperref[def:functor]{functor} \(F\colon \Set \to \Grp\) that sends a set \(X\) to the free group on \(X\).
    \item There is a \hyperref[def:functor]{functor} \(-^{\ast}\colon \Vect_{\mathbb{K}}^{\op} \to \Vect_{\mathbb{K}}\) that sends a vector space \(V\) to its dual \(V^{\ast}\).
    \item There is a \hyperref[def:functor]{functor} \(\operatorname{Spec}\colon \mathsf{CRing}^{\op} \to \Top\) that sends a commutative ring \(R\) to its set \(\operatorname{Spec}(R)\) of prime ideals endowed with the Zariski topology.
  \end{itemize}
\end{example}


\begin{example}[Identity functor]\label{ex:identity-functor}
  Let \(\category{C}\) be a \hyperref[def:category]{category}.
  There exists an identity \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\id\colon\)} \category{C} \arrow[r] & \category{C} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & X \arrow[d, "f" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & Y
    \end{tikzcd}
  \end{center}
\end{example}

\begin{example}[Constant functor]\label{ex:constant-functor}
  Let \(\category{C}\) be a \hyperref[def:category]{category}.
  For every object \(C \in \category{C}\) there exists a constant \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\operatorname{ct}_{C}\colon \;\;\)} \category{C} \arrow[r] & \category{C} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & C \arrow[d, "\id_{C}" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & C
    \end{tikzcd}
  \end{center}
\end{example}

\begin{example}[Bifunctor] \label{ex:bifunctor}
  Let \(\category{C}, \category{C}'\) and \(\category{D}\) be three \hyperref[def:category]{categories}.
  Recall that we can construct the \hyperref[def:product-category]{product category} \(\category{C} \times \category{C}'\).

  A \textbf{bifunctor} is a \hyperref[def:functor]{functor} of the form \(F\colon \category{C} \times \category{C}' \to \category{D}\).
  This means that for any \(X \in \category{C}\) and \(X' \in \category{C}\), \(F(X, -)\colon \category{C}' \to \category{D}\) and \(F(-, X')\colon \category{C} \to \category{D}\) are \hyperref[def:functor]{functors}.

  Furthermore for any morphisms \(f\colon X \to Y\) in \(\category{C}\) and \(g\colon X' \to Y'\) in \(\category{C}'\) we have a commutative diagram
  \begin{center}
    \begin{tikzcd}[row sep=huge, column sep=huge]
      F(X, X') \arrow[r, "{F(X, g)}"] \arrow[d, "{F(f, X')}"'] \arrow[dr, "{F(f, g)}"] & F(X, Y') \arrow[d, "{F(f, Y')}"] \\
      F(Y, X') \arrow[r, "{F(Y, g)}"'] & F(Y, Y')
    \end{tikzcd}
  \end{center}
\end{example}


\begin{example}\label{ex:hom-functor}
  Let \(\category{C}\) be a \hyperref[def:category]{category}.
  For every object \(C \in \category{C}\) there exists a \textit{covariant} \(\Hom\) \hyperref[def:functor]{functor}
   \begin{center}
    \begin{tikzcd}
      \llap{\(\Hom(C, -)\colon\)} \category{C} \arrow[r] & \Set \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & \Hom(C, X) \arrow[d, "f \circ " ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & \Hom(C, Y)
    \end{tikzcd}
  \end{center}
  and a \textit{contravariant} \(\Hom\) \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\Hom(-, C)\colon\)} \category{C} \arrow[r] & \Set \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & \Hom(C, X)  \arrow[from=source, to=target, mapsto] \\
      Y  & \Hom(C, Y) \arrow[u, "\circ f"' ""{name=target, left}]
    \end{tikzcd}
  \end{center}

  We can combine this two into a \hyperref[ex:bifunctor]{bifunctor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\text{Hom}(-,-)\colon\)} \opcategory{C} \times \category{C} \arrow[r] & \Set \\[-2em]
      (X, Y) \arrow[d, "{(f, g)}"' ""{name=source, right}]  & \text{Hom}(X, Y) \arrow[d, "{g \circ \,-\, \circ f}", ""{name=target, left}] \\
      (Z, W) & \text{Hom}(Z, W)
      \arrow[from=source, to=target, mapsto]
    \end{tikzcd}
  \end{center}
\end{example}

\begin{example}\label{ex:inclussion-functor}
  Let \(\category{D}\) be a \hyperref[def:subcategory]{subcategory} of a \hyperref[def:category]{category} \(\category{C}\).
  There exist an inclussion \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\iota\colon\)} \category{D} \arrow[r, hookrightarrow] & \category{C} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & X \arrow[d, "f" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & Y
    \end{tikzcd}
  \end{center}
\end{example}

\begin{example}[Preimage functor]
  Let \(X, Y\) be two topological spaces.
  Consider their \hyperref[ex:open-set-category]{categories of open sets} \(\Open(X), \Open(Y)\).

  For every continuous map \(f: X \to Y\) we have a preimage \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(f^{-1}\colon\)} \Open(Y) \arrow[r] & \Open(X) \\[-2em]
      U \arrow[d, ""{name=source, right}, hookrightarrow] & f^{-1}(U) \arrow[d, ""{name=target, left}, hookrightarrow] \arrow[from=source, to=target, mapsto] \\
      V  & f^{-1}(V)
    \end{tikzcd}
  \end{center}
\end{example}

\begin{proposition}[Functors preserve isomorphisms]
  Let \(\category{C}, \category{D}\) be two \hyperref[def:category]{categories} and \(F\colon \category{C} \to \category{D}\) a \hyperref[def:functor]{functor} between them.
  If \(f\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{C}\) then \(Ff\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{D}\).
\end{proposition}
\begin{proof}
  Let \(f\colon X \to Y\) be a \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{C}\).
  Then there exists an inverse \(f^{-1}\colon Y \to X\) in \(\category{C}\) such that
  \begin{equation*}
    f^{-1} \circ f = \id_{X} \quad f \circ f^{-1} = \id_{Y}
  \end{equation*}

  Applying \(F\) to this equalities we have
  \begin{align*}
    F(f^{-1} \circ f) = F\id_{X} &\implies Ff^{-1} \circ Ff = \id_{F(X)} \\
    F(f \circ f^{-1}) = F\id_{Y} &\implies Ff \circ Ff^{-1} = \id_{F(Y)}
  \end{align*}
  i.e: \(Ff\) has an inverse \(Ff^{-1}\) in \(\category{D}\).
\end{proof}


\begin{definition}[Functor types]\label{def:functor-types}
  A \hyperref[def:functor]{functor} \(F\colon \category{C} \to \category{D}\) is:
  \begin{itemize}
    \item \textbf{faithful} if for every pair of objects \(X, Y \in \category{C}\), the map \(\Hom_{\category{C}}(X, Y) \to \Hom_{\category{D}}(F(X), F(Y))\) is injective.
    \item \textbf{full} if for every pair \(X, Y \in \category{C}\), the map \(\Hom_{\category{C}}(X, Y) \to \Hom_{\category{D}}(F(X), F(Y))\) is surjective.
    \item \textbf{fully faithful} if it is both full and faithful.
    \item \textbf{essentially surjective} if for every object \(Y \in \category{D}\) there exists an object \(X \in \category{C}\) such that \(F(X) \cong Y\).
  \end{itemize}
\end{definition}

\begin{example}~
  \begin{itemize}
    \item The \hyperref[def:functor]{functor} \(U\colon \Grp \to \Set\) that sends a group to its underlying set is \hyperref[def:functor-types]{faithful}.
    \item A \hyperref[def:subcategory]{subcategory} \(\category{D}\) of a \hyperref[def:category]{category} \(\category{C}\) is \hyperref[def:subcategory]{full} if and only if the \hyperref[ex:inclussion-functor]{inclussion functor} \(\iota: \category{D} \to \category{C}\) is \hyperref[def:functor-types]{full}.
    \item Given a field \(\mathbb{K}\), let \(\category{C}_{\mathbb{K}}\) be the \hyperref[def:category]{category} whose objects are \(\mathbb{K}^{n}\) for some \(n \in \mathbb{N}\).
          The inclussion \(\category{C}_{\mathbb{K}} \to \Vect_{\mathbb{K}}\) is \hyperref[def:functor-types]{essentially surjective}, since all finite dimensional \(\mathbb{K}\)-vector spaces are isomorphic to \(\mathbb{K}^{n}\) for some \(n\).
  \end{itemize}
\end{example}


\section{Natural transformations}

\begin{definition}\label{def:natural-transformation}
  Let \(F, G\colon \category{C} \to \category{D}\) be two functors. A \textbf{natural transformation} \(\psi\colon F \implies G\) consists of a family of morphisms \(\psi_X: F(X) \to G(X)\) in \(\category{D}\), one for each object \(X \in \category{C}\), called the \textbf{components} of \(\psi\), such that for every morphism \(f\colon X \to Y\) in \(\category{C}\), the following diagram commutes:
  \begin{center}
    \begin{tikzcd}
      F(X) \arrow[r, "\psi_{X}"] \arrow[d, "Ff"'] & G(X) \arrow[d, "Gf"] \\
      F(Y) \arrow[r, "\psi_{Y}"'] & G(Y)
    \end{tikzcd}
  \end{center}
  Diagramatically, a natural transformations is represented by
  \begin{center}
    \begin{tikzcd}
      \category{C} \arrow[r, bend left=50, "F" ""{name=source, below}]
      \arrow[r, bend right=50, "G"' ""{name=target, above}]
      & \category{D}
      \arrow[Rightarrow, from=source, to=target, "\psi"]
    \end{tikzcd}
  \end{center}
  
  If each component \(\psi_{X}\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{D}\), we say that \(\psi\) is a \textbf{natural isomorphism}, and we write \(F \cong G\).
\end{definition}

\begin{example}
  In \(\Vect_{\mathbb{K}}\), the map \(\operatorname{ev}\colon V \to V^{\ast\ast}\) that sends a vector \(v \in V\) to the linear evaluation function \(\operatorname{ev}_{v}\colon V^{\ast} \to \mathbb{K}\) defines the components of a \hyperref[def:natural-transformation]{natural transformation} from the \hyperref[ex:identity-functor]{identity functor} \(\id\colon \Vect_{\mathbb{K}} \to \Vect_{\mathbb{K}}\) to the double dual \hyperref[def:functor]{functor} \(-^{\ast\ast}\colon \Vect_{\mathbb{K}} \to \Vect_{\mathbb{K}}\).

  If we restrict ourselves to the finite-dimensional case then this is a \hyperref[def:natural-transformation]{natural isomorphism}.
\end{example}

\begin{definition}[Equivalent categories]\label{def:equivalent-categories}
  Two \hyperref[def:category]{categories} \(\category{C}, \category{D}\) are \textbf{equivalent} if there exist \hyperref[def:functor]{functors} \(F\colon \category{C} \to \category{D}\) and \(G\colon \category{D} \to \category{C}\) such that \(G \circ F \cong \id_{\category{C}}\) and \(F \circ G \cong \id_{\category{D}}\). In this case, \(F\) (and \(G\)) is called an \textbf{equivalence of categories}.
  We denote this by \(\category{C} \simeq \category{D}\).
\end{definition}

In general proving that two \hyperref[def:equivalent-categories]{equivalence of categories} by directly constructing the \hyperref[def:functor]{functors} \(F\) and \(G\) is not easy.
Instead one can use the following theorem, which we state without proof.


\begin{theorem}[{\cite[Theorem 1.5.9]{riehlCategoryTheoryContext}}]\label{thm:equivalent-categories-characterization}
  A \hyperref[def:functor]{functor} \(F\colon \category{C} \to \category{D}\) defining an \hyperref[def:equivalent-categories]{equivalence of categories} is \hyperref[def:functor-types]{fully faithful and essentially surjective}.
  Assuming the axiom of choice, any \hyperref[def:functor]{functor} with these properties defines an  \hyperref[def:equivalent-categories]{equivalence of categories}.
\end{theorem}
