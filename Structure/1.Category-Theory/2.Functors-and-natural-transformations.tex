\chapter{Functors and natural transformations}

\section{Functors}

\begin{definition}[Functor]\label{def:functor}
  Let \(\category{C}, \category{D}\) be two \hyperref[def:big-category]{categories}.
  A (covariant) \textbf{functor} \(F: \category{C} \to \category{D}\) consists of:
  \begin{itemize}
    \item for each object \(X \in \category{C}\), an object \(F(X) \in \category{D}\).
    \item for each morphism \(f: X \to Y\) in \(\category{C}\), a morphism \(F(f): F(X) \to F(Y)\) in \(\category{D}\),
  \end{itemize}
  such that:
  \begin{itemize}
    \item \(F(\id_X) = \id_{F(X)}\) for every \(X \in \category{C}\).
    \item \(F(g \circ f) = F(g) \circ F(f)\) for every composable pair of morphisms \(f, g\) in \(\category{C}\).
  \end{itemize}

  A functor is \textbf{contravariant} if it reverses the order of composition: \(F(g \circ f) = F(f) \circ F(g)\). This is equivalent to a covariant functor \(F: \opcategory{C} \to \category{D}\).

  Diagramatically, a covariant functor is represent by
  \begin{center}
    \begin{tikzcd}
      \llap{\(F: \;\;\)} \category{C} \arrow[r] & \category{D} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & FX \arrow[d, "Ff" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & FY
    \end{tikzcd}
  \end{center}
  And a contravariant functor by
  \begin{center}
    \begin{tikzcd}
      \llap{\(F: \;\;\)} \category{C} \arrow[r] & \category{D} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & FX  \arrow[from=source, to=target, mapsto] \\
      Y  & FY \arrow[u, "Ff"' ""{name=target, left}]
    \end{tikzcd}
  \end{center}
\end{definition}

\begin{remark}
  We will disfavour the covariant/contravariant distintinction.
  Therefore all \hyperref[def:functor]{functors} will be covariant (unless otherwise specified) and contravariant \hyperref[def:functor]{functors} will be covariant \hyperref[def:functor]{functors} from the \hyperref[def:opposite-category]{opposite category}.
\end{remark}

\begin{example}[Identity functor]\label{ex:identity-functor}
  Let \(\category{C}\) be a \hyperref[def:big-category]{category}.
  There exists an identity \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\id: \;\;\)} \category{C} \arrow[r] & \category{C} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & X \arrow[d, "f" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & Y
    \end{tikzcd}
  \end{center}
\end{example}

\begin{example}[Constant functor]\label{ex:constant-functor}
  Let \(\category{C}\) be a \hyperref[def:big-category]{category}.
  For every object \(C \in \category{C}\) there exists a constant \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\operatorname{ct}_{C}: \;\;\)} \category{C} \arrow[r] & \category{C} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & C \arrow[d, "\id_{C}" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & C
    \end{tikzcd}
  \end{center}
\end{example}

\begin{example}\label{ex:hom-functor}
  Let \(\category{C}\) be a \hyperref{def:category}{locally small category}.
  For every object \(C \in \category{C}\) there exists a \textit{covariant} \(\Hom\) \hyperref[def:functor]{functor}
   \begin{center}
    \begin{tikzcd}
      \llap{\(\Hom(C, -): \;\;\)} \category{C} \arrow[r] & \Set \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & \Hom(C, X) \arrow[d, "f \circ " ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & \Hom(C, Y)
    \end{tikzcd}
  \end{center}
  and a \textit{contravariant} \(\Hom\) \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\Hom(-, C): \;\;\)} \category{C} \arrow[r] & \Set \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & \Hom(C, X)  \arrow[from=source, to=target, mapsto] \\
      Y  & \Hom(C, Y) \arrow[u, "\circ f"' ""{name=target, left}]
    \end{tikzcd}
  \end{center}

  We can combine this two into a \textit{bifunctor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\text{Hom}(-,-): \;\;\)} \opcategory{C} \times \category{C} \arrow[r] & \Set \\[-2em]
      (X, Y) \arrow[d, "{(f, g)}"' ""{name=source, right}]  & \text{Hom}(X, Y) \arrow[d, "{g \circ \,-\, \circ f}", ""{name=target, left}] \\
      (Z, W) & \text{Hom}(Z, W)
      \arrow[from=source, to=target, mapsto]
    \end{tikzcd}
  \end{center}
\end{example}

\begin{example}\label{ex:inclussion-functor}
  Let \(\category{D}\) be a \hyperref[def:subcategory]{subcategory} of a \hyperref[def:category]{category} \(\category{C}\).
  There exist an inclussion \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(\iota: \;\;\)} \category{D} \arrow[r, hookrightarrow] & \category{C} \\[-2em]
      X \arrow[d, "f"' ""{name=source, right}] & X \arrow[d, "f" ""{name=target, left}] \arrow[from=source, to=target, mapsto] \\
      Y  & Y
    \end{tikzcd}
  \end{center}
\end{example}


\begin{example}[Preimage functor]
  Let \(X, Y\) be two topological spaces.
  Consider their \hyperref[ex:open-set-category]{categories of open sets} \(\Open(X), \Open(Y)\).

  For every continuous map \(f: X \to Y\) we have a preimage \hyperref[def:functor]{functor}
  \begin{center}
    \begin{tikzcd}
      \llap{\(f^{-1}: \;\;\)} \Open(Y) \arrow[r] & \Open(X) \\[-2em]
      U \arrow[d, ""{name=source, right}, hookrightarrow] & f^{-1}(U) \arrow[d, ""{name=target, left}, hookrightarrow] \arrow[from=source, to=target, mapsto] \\
      V  & f^{-1}(V)
    \end{tikzcd}
  \end{center}
\end{example}

\begin{proposition}[Functors preseve isomorphisms]
  Let \(\category{C}, \category{D}\) be two \hyperref[def:big-category]{categories} and \(F: \category{C} \to \category{D}\) a \hyperref[def:functor]{functor} between them.
  If \(f\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{C}\) then \(Ff\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{D}\).
\end{proposition}

\begin{definition}[Functor types]\label{def:functor-types}
  A \hyperref[def:functor]{functor} \(F: \category{C} \to \category{D}\) is:
  \begin{itemize}
    \item \textbf{faithful} if for every pair of objects \(X, Y \in \category{C}\), the map \(\Hom_{\category{C}}(X, Y) \to \Hom_{\category{D}}(F(X), F(Y))\) is injective.
    \item \textbf{full} if for every pair \(X, Y \in \category{C}\), the map \(\Hom_{\category{C}}(X, Y) \to \Hom_{\category{D}}(F(X), F(Y))\) is surjective.
    \item \textbf{fully faithful} if it is both full and faithful.
    \item \textbf{essentially surjective} if for every object \(Y \in \category{D}\) there exists an object \(X \in \category{C}\) such that \(F(X) \cong Y\).
  \end{itemize}
\end{definition}

\begin{example}
  A \hyperref[def:subcategory]{subcategory} \(\category{D}\) of a \hyperref[def:category]{category} \(\category{C}\) is \hyperref[def:subcategory]{full} if and only if the \hyperref[ex:inclussion-functor]{inclussion functor} \(\iota: \category{D} \to \category{C}\) is \hyperref[def:functor-types]{full}.
\end{example}


\section{Natural transformations}

\begin{definition}\label{def:natural-transformation}
  Let \(F, G: \category{C} \to \category{D}\) be two functors. A \textbf{natural transformation} \(\psi: F \implies G\) consists of a family of morphisms \(\psi_X: F(X) \to G(X)\) in \(\category{D}\) for each object \(X \in \category{C}\), called the \textbf{components} of \(\psi\), such that for every morphism \(f: X \to Y\) in \(\category{C}\), the following diagram commutes:
  \begin{center}
    \begin{tikzcd}
      F(X) \arrow[r, "\psi_X"] \arrow[d, "F(f)"'] & G(X) \arrow[d, "G(f)"] \\
      F(Y) \arrow[r, "\psi_Y"'] & G(Y)
    \end{tikzcd}
  \end{center}
  Diagramatically, a natural transformations is represented by
  \begin{center}
    \begin{tikzcd}
      \category{C} \arrow[r, bend left=50, "F" ""{name=source, below}]
      \arrow[r, bend right=50, "G"' ""{name=target, above}]
      & \category{D}
      \arrow[Rightarrow, from=source, to=target, "\psi"]
    \end{tikzcd}
  \end{center}
  
  If each component \(\psi_{X}\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{D}\), we say that \(\psi\) is a \textbf{natural isomorphism}, and we write \(F \cong G\).
\end{definition}

\begin{definition}[Composition of natural transformations]~
  \begin{itemize}
    \item \textbf{Vertical composition}: Given natural transformations \(\eta: F \implies G\) and \(\varepsilon: G \implies H\), their vertical composite \(\varepsilon \cdot \eta: F \implies H\) is defined component-wise by \((\varepsilon \cdot \eta)_X = \varepsilon_X \circ \eta_X\).
    \item \textbf{Horizontal composition}: Given natural transformations \(\eta: F \implies G\) (where \(F, G: \category{C} \to \category{D}\)) and \(\mu: H \implies K\) (where \(H, K: \category{D} \to \category{E}\)), their horizontal composite \(\mu * \eta: H \circ F \implies K \circ G\) is defined by \((\mu * \eta)_X = \mu_{G(X)} \circ H(\eta_X) = K(\eta_X) \circ \mu_{F(X)}\).
  \end{itemize}
\end{definition}

\begin{definition}[Equivalent categories]\label{def:equivalent-categories}
  Two categories \(\category{C}, \category{D}\) are \textbf{equivalent} if there exist functors \(F: \category{C} \to \category{D}\) and \(G: \category{D} \to \category{C}\) such that \(G \circ F \cong \id_{\category{C}}\) and \(F \circ G \cong \id_{\category{D}}\). In this case, \(F\) (and \(G\)) is called an \textbf{equivalence of categories}.
\end{definition}

\begin{theorem}
  A functor \(F: \category{C} \to \category{D}\) is an \hyperref[def:equivalent-categories]{equivalence of categories} if and only if it is \hyperref[def:functor-types]{fully faithful and essentially surjective}.
\end{theorem}
