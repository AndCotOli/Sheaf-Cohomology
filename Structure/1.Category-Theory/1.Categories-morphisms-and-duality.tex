\chapter{Categories, morphisms and duality}

\section{Categories}

\begin{definition}[Category]\label{def:category}
  A \textbf{category} \(\category{C}\) consists of:
  \begin{itemize}
    \item a class \(\operatorname{Ob}(\category{C})\) of \textbf{objects}.
    \item for each pair of objects \(X, Y \in \operatorname{Ob}(\category{C})\) a collection of sets \(\Hom_{\category{C}}(X, Y)\) of \textbf{morphisms from \(X\) to \(Y\)}.
    \item for each object \(X \in \operatorname{Ob}(\category{C})\) a morphisms \(\id_{X} \in \Hom_{\category{C}}(X, X)\) called the \textbf{identity} morphism.
    \item for each pair of morphisms \(f \in \Hom_{\category{C}}(X, Y)\) and \(g \in \Hom_{\category{C}}(Y, Z)\) a morphism \(g \circ f \in \Hom_{\category{C}}(X, Z)\) called their \textbf{composite}.
  \end{itemize}
  satisfying the following axioms:
  \begin{itemize}
    \item Composition is associative: for each quadruple \(X, Y, Z, W \in \operatorname{Ob}(\category{C})\) of objects, if \(f \in \Hom_{\category{C}}(Y, Z)\), \(g \in \Hom_{\category{C}}(X, Y)\) and \(h \in \Hom_{\category{C}}(W, X)\) then
          \begin{equation*}
            (f \circ g) \circ h = f \circ (g \circ h)
          \end{equation*}
    \item Composition satiesfies the unit laws: for each pair of objects \(X, Y \in \operatorname{Ob}(\category{C})\) if \(f \in \Hom_{\category{C}}(X, Y)\) then
          \begin{equation*}
            \id_{Y} \circ f = f = f \circ \id_{X}
          \end{equation*}
  \end{itemize}
\end{definition}

\begin{definition}[Small category]\label{def:small-category}
  A \hyperref[def:category]{category} \(\category{C}\) is \textbf{small} if its collection of objects \(\operatorname{Ob}(\category{C})\) is a set.
\end{definition}

\begin{remark}[Size issues]
  We will be sketchy about foundational issues (set-theoretic issues, universes, ...).
  
  Therefore all \hyperref[def:category]{categories} will be \(\mathcal{U}\)-categories for a large enough \textit{universe} \(\mathcal{U}\) (i.e: for any pair of objects \(X, Y \in \category{C}\) the \(\Hom\)-set \(\Hom(X, Y)\) is \(\mathcal{U}\)-small) and \hyperref[def:small-category]{small categories} will be \(\mathcal{U}\)-small (the set of objects \(\operatorname{Ob}(\category{C})\) is \(\mathcal{U}\)-small).
  
  We will refer as "big" categories those that are \(\mathcal{V}\)-categories for a larger universe \(\mathcal{V}\).
  By Grothendieck's axiom (for any set \(X\) there exists a universe \(\mathcal{U}\) such that \(X \in \mathcal{U}\)) every category \(\category{C}\) is a \(\mathcal{V}\)-category for a big enough universe \(\mathcal{V}\), and this universe can be chosen so that \(\category{C}\) is \(\mathcal{V}\)-small.
  
  For a basic introduction of what an \textit{universe} is and a more carefull treatment of this type of size issues see \cite[Section 1.1]{kashiwaraCategoriesSheaves2006}.
  More details can also be found at \cite{SGA4}
\end{remark}

\begin{notation}
  To indicate that \(X\) is an object of a \hyperref[def:category]{category} \(\category{C}\) we write
  \begin{equation*}
    X \in \category{C} \text{ instead of } X \in \operatorname{Ob}(\category{C})
  \end{equation*}
  To indicate that \(f \in \Hom_{\category{C}}(X, Y)\) is a morphism from \(X\) to \(Y\) we write
  \begin{equation*}
    f\colon X \to Y \text{ or } X \xrightarrow{f} Y
  \end{equation*}
  We will sometimes abbreviate the composition \(f \circ g\) as \(fg\).
\end{notation}


\begin{example}~
  \begin{itemize}
    \item The \hyperref[def:category]{category} \(\Set\) has the collection of all sets as objects and mappings as morphisms.
    \item The \hyperref[def:category]{category} \(\Grp\) has groups as objects and group homomorphisms as morphisms.
    \item The \hyperref[def:category]{category} \(\Ab\) has abelian groups as objects and group homomorphisms as morphisms. It is a full subcategory of \(\Grp\).
    \item The \hyperref[def:category]{category} \(\Ring\) has rings as objects and ring homomorphisms as morphisms.\footnote{We adopt the convetion that all rings have an unit and all ring homomorphisms preserve it.}
    \item The \hyperref[def:category]{category} \(\category{Rng}\) has (not necessarilly unital) rings as objects and ring homomorphisms as morphism.
    \item The \hyperref[def:category]{category} \(\Top\) has topological spaces as objects and continuous functions as morphisms.
    \item For a fixed ring \(R\), the \hyperref[def:category]{category} \(\Mod(R)\) has (left) \(R\)-modules as objects and \(R\)-linear maps as morphisms.
    \item For a fixed field \(k\), the \hyperref[def:category]{category} \(\Vect_{k}\) has \(k\)-vector spaces as objects and \(k\)-linear maps as morphisms.
  \end{itemize}
\end{example}

\begin{example}[Category of open sets]\label{ex:open-set-category}
  Let \(X\) be a topological space.
  There is a \hyperref[def:category]{category} of open sets \(\Open(X)\) consisting of
  \begin{itemize}
    \item As objects, the open subsets of \(X\).
    \item As morphisms, inclusion of open sets: \(U \to V\) if and only if \(U \subseteq V\)
  \end{itemize}
\end{example}

\begin{example}[Discrete category]\label{ex:discrete-category}
  Given a \hyperref[def:category]{category} \(\category{C}\) we can define the \textbf{discrete category} \(\category{C}_{d}\) taking the same collection of objects as \(\category{C}\) but only the identity morphisms between them.
\end{example}

\begin{definition}[Subcategory]\label{def:subcategory}
  Given a \hyperref[def:category]{category} \(\category{C}\), a \textbf{subcategory} \(\category{D}\) consists of a subcollection of objects and a subcollection of morphisms in \(\category{C}\) such that
  \begin{itemize}
    \item If \(f\colon X \to Y\) is a morphism in \(\category{D}\) then \(X, Y\) are objects in \(\category{D}\).
    \item If \(f\colon X \to Y\) and \(g\colon Y \to Z\) are morphisms in \(\category{D}\), the composition \(gf\colon X \to Z\) is in \(\category{D}\).
    \item If \(X\) is an object in \(\category{D}\) then the identity map \(\id_{X}\) is in \(\category{D}\).
  \end{itemize}

  We say that \(\category{D}\) is a \textbf{full subcategory} if for all objects \(X, Y\) in \(\category{D}\) every morphism \(f\colon X \to Y\) in \(\category{C}\) is also in \(\category{D}\).
\end{definition}

\begin{example}~
  \begin{itemize}
    \item \(\Ab\) is a \hyperref[def:subcategory]{full subcategory} of \(\Grp\).
    \item The \hyperref[def:category]{category} of finite sets \(\category{FinSet}\) is a \hyperref[def:subcategory]{full subcategory} of \(\Set\).
    \item \(\Ring\) is a \hyperref[def:subcategory]{subcategory} of \(\category{Rng}\) that is not full.
    \item If \(k\) is a field, \(\Vect_{k}\) is a \hyperref[def:subcategory]{full subcategory} of \(\Mod(k)\).
  \end{itemize}
\end{example}

\begin{definition}[Product category]\label{def:product-category}
  Let \(\category{C}, \category{D}\) be two \hyperref[def:category]{categories}.
  The \textbf{product category} \(\category{C} \times \category{D}\) is the \hyperref[def:category]{category} consisting of
  \begin{itemize}
    \item As objects: ordered pairs \((C, D)\), where \(C\) is an object in \(\category{C}\) and \(D\) is an object in \(\category{D}\), i.e:
          \begin{equation*}
            \operatorname{Ob}(\category{C} \times \category{D}) \coloneq \operatorname{Ob}(\category{C}) \times \operatorname{Ob}(\category{D})
          \end{equation*}
    \item As morphisms: ordered pairs \((C \xrightarrow{f} C', D \xrightarrow{g} D')\), where \(C \xrightarrow{f} C'\) is a morphism in \(\category{C}\) and \(D \xrightarrow{g} D'\) is a morphism in \(\category{D}\), i.e: for all objects \((C, D), (C', D')\) in \(\category{C} \times \category{D}\)
          \begin{equation*}
            \Hom_{\category{C} \times \category{D}}((C, D), (C', D')) \coloneq \Hom_{\category{C}}(C, C') \times \Hom_{\category{D}}(D, D')
          \end{equation*}
    \item Composition is defined \textit{componentwise}: if \((C \xrightarrow{f} C', D \xrightarrow{g} D')\) and \((C' \xrightarrow{f'} C'', D' \xrightarrow{g'} D'')\) are two morphisms in \(\category{C} \times \category{D}\) then
          \begin{equation*}
            (f', g') \circ (f, g) \coloneq (f' \circ f, g' \circ g): C \times D \to C'' \times D''
          \end{equation*}
  \end{itemize}
\end{definition}

\subsection{Duality}

\begin{definition}[Opposite category]\label{def:opposite-category}
  Let \(\category{C}\) be a \hyperref[def:category]{category}. The \textbf{opposite category} \(\opcategory{C}\) consists of:
  \begin{itemize}
    \item The same objects as \(\category{C}\): \(\operatorname{Ob}(\opcategory{C}) = \operatorname{Ob}(\category{C})\).
    \item For each pair of objects \(X, Y \in \category{C}^{\op}\): \(\Hom_{\opcategory{C}}(X, Y) = \Hom_{\category{C}}(Y, X)\).
    \item If \(f^{\op}\colon X \to Y\), \(g^{\op}\colon Y \to Z\) are morphisms in \(\category{C}^{\op}\) their composition is defined by
          \begin{equation*}
            f^{\op} \circ g^{\op} = (g \circ f)^{\op}
          \end{equation*}
          where \(f\colon Y \to X\) and \(g\colon Z \to Y\) are the associated morphisms in \(\category{C}\).
          Visually
          \begin{center}
            \begin{tikzcd}
              Z \arrow[r, shift left, "g"] \arrow[rr, bend left=4em, "g \circ f"] & Y \arrow[r, shift left, "f"] \arrow[l, shift left, "g^{\op}"] & X \arrow[l, shift left, "f^{\op}"] \arrow[ll, dotted, bend left=4em, "f^{\op} \circ g^{\op}"]
            \end{tikzcd}
          \end{center}
  \end{itemize}
\end{definition}

\begin{example}\label{ex:poset-category-dual}
  Consider a preordered set \((P, \leq)\) viewed as a \hyperref[def:small-category]{small category} where a unique morphism \(x \to y\) exists if and only if \(x \leq y\).

  The \hyperref[def:opposite-category]{opposite category} \(P^{\op}\) corresponds to the same set \(P\) equipped with the reversed ordering \(\geq\). That is, a morphism \(x \to y\) exists in \(P^{\text{op}}\) if and only if \(y \leq x\) in \(P\).
\end{example}

\begin{remark}
  Let \(f^{\op}\colon Y \to X\) and \(g^{\op}\colon Y \to X\) be two morphisms in the \hyperref[def:opposite-category]{opposite category} \(\opcategory{C}\).
  Then \(f^{\op} = g^{\op}\) in \(\opcategory{C}\) if and only if \(f = g\) in \(\category{C}\).
\end{remark}

A categorical statement \(P^{\ast}\) is defined as the \textbf{dual} of a statement \(P\) in a \hyperref[def:category]{category} \(\category{C}\) if \(P^{\ast}\) holds in the \hyperref[def:opposite-category]{opposite category} \(\opcategory{C}\) if and only if \(P\) holds in \(\category{C}\).

Note that, by definition, for any \hyperref[def:category]{category} \(\category{C}\) we have that \((\opcategory{C})^{\op} = \category{C}\).

Therefore if a statement is true for all \hyperref[def:category]{categories} \(\category{C}\), its dual is true for all \hyperref[def:category]{categories} \(\opcategory{C}\).
Since \(\opcategory{C}\) is an arbitrary \hyperref[def:category]{category} the dual statement is also true for all \hyperref[def:category]{categories}.

In practice, this means that proving an statement directly proves its dual for free.

\section{Types of morphisms}

\begin{definition}[Monomorphism and epimorphism]\label{def:mono-epi-morphism}
  Let \(f\colon X \to Y\) be a morphism in a \hyperref[def:category]{category} \(\category{C}\).
  We say that \(f\) is a
  \begin{itemize}
    \item \textbf{Monomorphism} (or is \textbf{monic}) if for every pair of parallel morphisms \(g, h\colon Z \to X\) in \(\category{C}\), the equality \(f \circ g = f \circ h\) implies \(g = h\), i.e: \(f\) is left-cancellative.
    \item \textbf{Epimorphism} (or is \textbf{epic}) if for every pair of parallel morphisms \(g, h\colon Y \to Z\) in \(\category{C}\), the equality \(g \circ f = h \circ f\) implies \(g = h\), i.e: \(f\) is right-cancellative.
  \end{itemize}
\end{definition}

\begin{example}~
  \begin{itemize}
    \item In \(\Set\), an \hyperref[def:mono-epi-morphism]{monomorphism} (resp. \hyperref[def:mono-epi-morphism]{epimorphism}) is an injective (resp. surjective) function.
    \item In \(\Grp\), an \hyperref[def:mono-epi-morphism]{monomorphism} (resp. \hyperref[def:mono-epi-morphism]{epimorphism}) is an injective (resp. surjective) group homomorphism.
    \item In \(\Top\), an \hyperref[def:mono-epi-morphism]{monomorphism} (resp. \hyperref[def:mono-epi-morphism]{epimorphism}) is an injective (resp. surjective) continuous map.
  \end{itemize}
\end{example}

\begin{example}
  In \(\Ring\), there are \hyperref[def:mono-epi-morphism]{epimorphisms} that are not surjective ring homomorphisms.

  For example, consider the inclussion \(\iota\colon \mathbb{Z} \hookrightarrow \mathbb{Q}\).
  For any ring homomorphism \(f\colon \mathbb{Q} \to R\), where \(R\) is any ring, since for all \(\frac{a}{b} \in \mathbb{Q}\) we have \(a, b \in \mathbb{Z}\)
  \begin{equation*}
    f \left( \frac{a}{b} \right) = f(a) f(b^{-1}) = f(a) f(b)^{-1}
  \end{equation*}
  Therefore \(f\) is completely determined by its values in \(\mathbb{Z}\) and, since \(f \circ \iota = f|_{\mathbb{Z}}\) we get that \(\iota\) is right-cancellative.
\end{example}


\begin{lemma}[Monomorphisms and epimorphisms are dual]
  Let \(f\colon X \to Y\) be a morphism in a \hyperref[def:category]{category} \(\category{C}\).
  Consider the associated morphism \(f^{\op}\colon Y \to X\) in the \hyperref[def:opposite-category]{opposite category} \(\opcategory{C}\).

  Then \(f\) is a \hyperref[def:mono-epi-morphism]{monomorphism} (resp. \hyperref[def:mono-epi-morphism]{epimorphism}) in \(\category{C}\) if and only if \(f^{\op}\) is an \hyperref[def:mono-epi-morphism]{epimorphism} (resp. \hyperref[def:mono-epi-morphism]{epimorphism}) in \(\opcategory{C}\).
\end{lemma}
\begin{proof}
  Consider of pair of morphisms \(g, h\colon Z \rightrightarrows X\) in \(\category{C}\) and their opposites \(g^{\op}, h^{\op}\colon X \rightrightarrows Z\) in \(\category{C}^{\op}\).

  Note that, by definition of composition in \(\category{C}^{\op}\) we have
  \begin{equation*}
    g^{\op} \circ f^{\op} = (f \circ g)^{\op}
  \end{equation*}

  Therefore we have that \(f \circ g = f \circ h\) implies \(g = h\) if and only if \((f \circ g)^{\op} = (f \circ h)^{\op}\) implies \(g = h\) if and only if \(g^{\op} \circ f^{\op} = h^{\op} \circ f^{\op}\) implies \(g^{\op} = h^{\op}\).
\end{proof}

\begin{definition}[Isomorphism]\label{def:categorical-isomorphism}
  A morphism \(f\colon X \to Y\) in a \hyperref[def:category]{category} \(\category{C}\) is an \textbf{isomorphism} if there exists another morphism \(g\colon Y \to X\) in \(\category{C}\) such that
  \begin{equation*}
    g \circ f = \id_X \quad \text{and} \quad f \circ g = \id_Y
  \end{equation*}
  
  The morphism \(g\) is called the \textbf{inverse} of \(f\) and is denoted by \(f^{-1}\).
\end{definition}

\begin{proposition}
  Let \(f\colon X \to Y\) and \(g\colon Y \to Z\) be \hyperref[def:categorical-isomorphism]{isomorphisms} in a \hyperref[def:category]{category} \(\category{C}\). Then:
  \begin{enumerate}
    \item The inverse \(f^{-1}\) of \(f\) is unique.
    \item The identity \(\id_X\) is an \hyperref[def:categorical-isomorphism]{isomorphism} for every \(X \in \category{C}\).
    \item \(g \circ f\) is an \hyperref[def:categorical-isomorphism]{isomorphism} with \((g \circ f)^{-1} = f^{-1} \circ g^{-1}\).
    \item If \(g \circ f\) is an \hyperref[def:categorical-isomorphism]{isomorphism} and \(f\) is an \hyperref[def:categorical-isomorphism]{isomorphism}, then \(g\) is an \hyperref[def:categorical-isomorphism]{isomorphism}. Analogously if \(g \circ f\) and \(g\) are \hyperref[def:categorical-isomorphism]{isomorphisms}, then \(f\) is an \hyperref[def:categorical-isomorphism]{isomorphism}.
  \end{enumerate}
\end{proposition}
\begin{proof}~
  \begin{enumerate}
    \item Suppose that \(h_{1}, h_{2}: Y \to X\) are inverses of \(f\).
          We have
          \begin{equation*}
            h_{1} = h_{1} \circ \id_{Y} = h_{1} \circ (f \circ h_{2}) = (h_{1} \circ f) \circ h_{2} = \id_{X} \circ h_{2} = h_{2}
          \end{equation*}
    \item For all \(X \in \category{C}\) by definition
          \begin{equation*}
            \id_{X} \circ \id_{X} = \id_{X}
          \end{equation*}
          i.e: \(\id_{X}\) is its own left and right inverse.
    \item We only need to check that \(f^{-1} \circ g^{-1}\) is both a left and right inverse of \(g \circ f\).
          On the one hand
          \begin{equation*}
            (f^{-1} \circ g^{-1}) \circ (g \circ f) = f^{-1} \circ (g^{-1} \circ g) \circ f = f^{-1} \circ f = \id_{X}
          \end{equation*}
          On the other
          \begin{equation*}
            (g \circ f) \circ (f^{-1} \circ g^{-1}) = g \circ (f \circ f^{-1}) \circ g^{-1} = g \circ g^{-1} = \id_{Z}
          \end{equation*}
    \item Let \(h \coloneq (g \circ f)^{-1}: Z \to X\).
          We are going to check that \(f \circ h: Z \to Y\) is the inverse of \(g\).
          On the one hand
          \begin{equation*}
            g \circ (f \circ h) = (g \circ f) \circ h = \id_{Z}
          \end{equation*}
          On the other, since \(h \circ (g \circ f) = \id_{X}\) precomposing with \(f\) and postcomposing with \(f^{-1}\) we have
          \begin{align*}
            f \circ (h \circ g \circ f) \circ f^{-1} = f \circ \id_{X} \circ f^{-1} = \id_{Y} &\implies (f \circ h) \circ g \circ (f \circ f^{-1}) = \id_{Y} \\
                                                                      &\implies (f \circ h) \circ g = \id_{Y} \qedhere
          \end{align*}
  \end{enumerate}
\end{proof}

\begin{example}~
  \begin{itemize}
    \item In \(\Set\), an \hyperref[def:categorical-isomorphism]{isomorphism} is a bijection.
    \item In \(\Grp\), an \hyperref[def:categorical-isomorphism]{isomorphism} is a bijective group homomorphism.
    \item In \(\Top\), an \hyperref[def:categorical-isomorphism]{isomorphism} is a homeomorphism.
    \item In \(\Ring\), an \hyperref[def:categorical-isomorphism]{isomorphism} is a bijective ring homomorphism.
  \end{itemize}
\end{example}

\begin{proposition}
  Let \(f\colon X \to Y\) be an \hyperref[def:categorical-isomorphism]{isomorphism}.
  Then \(f\) is both a \hyperref[def:mono-epi-morphism]{monomorphism and epimorphism}.
\end{proposition}
\begin{proof}
  Suppose \(f\colon X \to Y\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in a \hyperref[def:category]{category} \(\category{C}\).
  Then there exists a morphism \(f^{-1}\colon Y \to X\) in \(\category{C}\) such that
  \begin{equation*}
    f^{-1} \circ f = \id_{X} \quad f \circ f^{-1} = \id_{Y}
  \end{equation*}

  Consider a pair of morphisms \(g, h\colon Z \rightrightarrows X\) such that \(f \circ g = f \circ h\).
  Precomposing with \(f^{-1}\) we get
  \begin{align*}
    f^{-1} \circ (f \circ g) = f^{-1} \circ (f \circ h) \implies (f^{-1} \circ f) \circ g = (f^{-1} \circ f) \circ h \implies g = h
  \end{align*}
  Therefore \(f\) is a \hyperref[def:mono-epi-morphism]{monomorphism}.

  Analogously, if we consider pair of morphism \(g', h'\colon Y \rightrightarrows Z\) such that \(g' \circ f = h' \circ f\) this time by postcomposing with \(f^{-1}\) we get \(g' = h'\), i.e: \(f\) is a \hyperref[def:mono-epi-morphism]{epimorphism}.
\end{proof}

\begin{example}
  In general, a morphism \(f\colon X \to Y\) that is both a \hyperref[def:mono-epi-morphism]{monomorphism and epimorphism} is not an \hyperref[def:categorical-isomorphism]{isomorphism}.

  Indeed, in the \hyperref[def:category]{category} \(\Top\) of topological spaces consider a set with more than one point \(X\) endowed with the discrete \(X_{d}\) and trivial \(X_{t}\) topology.
  
  Consider the identity map \(\id_{X}\colon X_{d} \to X_{t}\), which is continuous.
  Since \(\id_{X}\) is injective we have that it is a \hyperref[def:mono-epi-morphism]{monomorphism} and, since it is surjective, it is an \hyperref[def:mono-epi-morphism]{epimorphism}.

  However since there does not exists a continuous bijection \(X_{t} \to X_{d}\), \(\id_{X}\) cannot be an \hyperref[def:categorical-isomorphism]{isomorphism}.
\end{example}


\begin{proposition}
  Let \(\category{D}\) be a \hyperref[def:subcategory]{subcategory} of a \hyperref[def:category]{category} \(\category{C}\).
  If \(f\colon X \to Y\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{D}\) then \(f\) is also an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{C}\).
\end{proposition}
\begin{proof}
  Since \(f\colon X \to Y\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{D}\) there exists an inverse \(f^{-1}\colon Y \to X\) in \(\category{D}\).
  Therefore since \(\Hom_{\category{D}}(Y, X) \subseteq \Hom_{\category{C}}(Y, X)\), \(f^{-1}\) is an inverse of \(f\) in \(\category{C}\).
\end{proof}


\begin{proposition}\label{prop:full-subcategory-reflects-isomorphisms}
  Let \(\category{D}\) be a \hyperref[def:subcategory]{full subcategory} of a \hyperref[def:category]{category} \(\category{C}\).
  If \(f\colon X \to Y\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{C}\) and \(X, Y \in \category{D}\) then \(f\) is also an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{D}\).
\end{proposition}
\begin{proof}
  Since \(f\colon X \to Y\) is an \hyperref[def:categorical-isomorphism]{isomorphism} in \(\category{C}\) there exists an inverse \(f^{-1}\colon Y \to X\) in \(\category{D}\).
  Since \(\category{D}\) is a \hyperref[def:subcategory]{full subcategory} of \(\category{D}\) and \(X, Y \in \category{D}\) we have that \(f^{-1} \in \Hom_{\category{D}}(Y, X)\).
  Therefore \(f\) admits an inverse in \(\category{D}\).
\end{proof}


\section{Special objects}

\begin{definition}[Initial and terminal objects]\label{def:initial-terminal-object}
  An object \(I \in \category{C}\) is \textbf{initial} if for every object \(X \in \category{C}\) there is a unique morphism \(I \to X\).

  Dually, an object \(T \in \category{C}\) is \textbf{terminal} if for every object \(X \in \category{C}\) there is a unique morphism \(X \to T\).
\end{definition}

\begin{example}~
  \begin{itemize}
    \item In \(\Set\), the empty set \(\varnothing\) is \hyperref[def:initial-terminal-object]{initial} and the singleton \(\{\ast\}\) is \hyperref[def:initial-terminal-object]{terminal}.
    \item In \(\Grp\), trivial groups are both \hyperref[def:initial-terminal-object]{initial} and \hyperref[def:initial-terminal-object]{terminal}.
    \item In \(\Ring\) both \(\mathsf{Rng}\), the zero ring is \hyperref[def:initial-terminal-object]{initial}.
          However, the \hyperref[def:initial-terminal-object]{terminal} is different: in \(\Ring\) it is \(\mathbb{Z}\) and in \(\mathsf{Rng}\) is the zero ring.
    \item In \(\Top\), the empty topological space is \hyperref[def:initial-terminal-object]{initial} and the singleton \(\{\ast\}\) is \hyperref[def:initial-terminal-object]{terminal}.
    \item There are no \hyperref[def:initial-terminal-object]{initial} nor \hyperref[def:initial-terminal-object]{terminal} objects, since there are no homomorphisms between fields of different characteristic.
  \end{itemize}
\end{example}

\begin{proposition}[Uniqueness of initial and terminal objects]\label{prop:uniqueness-initial-terminal-objects}
  Let \(X, Y\) be two \hyperref[def:initial-terminal-object]{initial} (resp. \hyperref[def:initial-terminal-object]{terminal}) objects in a \hyperref[def:category]{category} \(\category{C}\).
  Then there exists a unique \hyperref[def:categorical-isomorphism]{isomorphism} \(X \isoarrow Y\) in \(\category{C}\).
\end{proposition}
\begin{proof}
  Since \(X\) is \hyperref[def:initial-terminal-object]{initial} in \(\category{C}\) there exists a unique morphism \(f\colon X \to Y\).
  Analogously there exist a unique morphism \(g\colon Y \to X\).

  Note that \(g \circ f\colon X \to X\).
  Therefore since \(X\) is \hyperref[def:initial-terminal-object]{initial} \(g \circ f = \id_{X}\).
  Analogously since \(f \circ g: Y \to Y\) we have \(f \circ g = \id_{Y}\).

  Therefore \(f\) and \(g\) are inverses of each other, i.e: \(X \cong Y\).
  Moreover they are unique by definition.
\end{proof}

\begin{definition}[Zero object]\label{def:zero-object}
  An object \(0 \in \category{C}\) is a \textbf{zero} object if it is both \hyperref[def:initial-terminal-object]{initial} and \hyperref[def:initial-terminal-object]{terminal}.

  Note that for any two objects \(X, Y \in \category{C}\) there exist unique maps \(0_{X}\colon X \to 0\) and \(0_{Y}\colon 0 \to Y\).
  We call this composite map the \textbf{zero map} between \(X\) and \(Y\) and denote it \(X \xrightarrow{0_{X, Y}} Y\):
  \begin{center}
    \begin{tikzcd}
      X \arrow[dr, "0_{X}"'] \arrow[rr, dotted, "0_{X, Y}"] & & Y \\
      & 0 \arrow[ur, "0_{Y}"']
    \end{tikzcd}
  \end{center}

  By abuse of notation, we will denote all of these maps by \(0\).
\end{definition}

\begin{example}~
  \begin{itemize}
    \item \(\Set\) and \(\Top\) have no \hyperref[def:zero-object]{zero objects} since \(\varnothing \neq \{\ast\}\).
    \item In \(\Grp\) and \(\Ab\), the zero group is the \hyperref[def:zero-object]{zero object}.
    \item \(\Ring\) has no \hyperref[def:zero-object]{zero objects}. However in \(\mathsf{Rng}\) the zero ring is the \hyperref[def:zero-object]{zero object}.
  \end{itemize}
\end{example}

\begin{corollary}[Uniqueness of zero objects]\label{def:uniqueness-zero-objects}
  Let \(0, \widetilde{0}\) be two \hyperref[def:zero-object]{zero} objects in a \hyperref[def:category]{category} \(\category{C}\).
  Then there exists a unique \hyperref[def:categorical-isomorphism]{isomorphism} \(0 \isoarrow \widetilde{0}\) in \(\category{C}\).
\end{corollary}
\begin{proof}
  Since \(0\) and \(\widetilde{0}\) are in particular \hyperref[def:initial-terminal-object]{initial} we can apply \autoref{prop:uniqueness-initial-terminal-objects}.
\end{proof}
